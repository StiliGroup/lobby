function doGet() {
  return json_({ ok: true, servicio: 'Registro de Visitantes Stili', version: VERSION_SERVICIO, hora: new Date() });
}

function doPost(e) {
  let req;
  try {
    req = JSON.parse(e.postData.contents);
  } catch (err) {
    return json_({ ok: false, error: 'Solicitud inválida.' });
  }
  try {
    reiniciarCache_();
    const publica = typeof ACCIONES_CENTRAL !== 'undefined' ? ACCIONES_CENTRAL[req.accion] : null;
    if (publica) return json_(Object.assign({ ok: true }, publica(req)));
    const lobby = autenticar_(req.token);
    const accion = ACCIONES[req.accion];
    if (!accion) throw new Error('Acción desconocida: ' + req.accion);
    return json_(Object.assign({ ok: true }, accion(req, lobby)));
  } catch (err) {
    return json_({ ok: false, error: err.message, codigo: err.codigo || '' });
  }
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function error_(mensaje, codigo) {
  const e = new Error(mensaje);
  e.codigo = codigo || '';
  return e;
}

function accionInicio_(req, lobby) {
  const cfg = config_();
  const categorias = leer_('Categorias').filas
    .filter(c => !esNo_(c.activo))
    .map(c => ({
      codigo: s_(c.codigo), nombre: s_(c.nombre), letra: s_(c.letra) || s_(c.codigo).charAt(0),
      horas_max: Number(c.horas_max) || 0, acceso: s_(c.acceso), requiere_destino: esSi_(c.requiere_destino),
      es_personal_trabajo: esSi_(c.es_personal_trabajo), color: s_(c.color) || '#1F3A5F',
    }));
  const apartamentos = leer_('Apartamentos').filas
    .filter(a => mismaTorre_(a.torre, lobby.torre) && !esNo_(a.activo))
    .map(a => ({ apartamento: s_(a.apartamento), residente: s_(a.residente), instrucciones: s_(a.instrucciones), ocupacion_maxima: Number(a.ocupacion_maxima) || 0 }));
  return {
    lobby: lobby,
    empresa: cfg.NOMBRE_EMPRESA || 'Stili Group',
    aviso_minutos_cierre: Number(cfg.AVISO_MINUTOS_CIERRE) || 15,
    categorias: categorias,
    apartamentos: apartamentos,
    tipos_incidencia: TIPOS_INCIDENCIA,
    cierres_hoy: cierresHoy_(lobby.torre, categorias),
    dentro: dentro_(lobby.torre),
    hoy: resumenHoy_(lobby.torre),
    capacidad: alertasCapacidad_(lobby.torre),
    registro_individual: !esNo_(cfg.REGISTRO_INDIVIDUAL),
  };
}

/** Perfil del visitante: datos, visitas anteriores, incidencias y lista negra. */
function accionPerfil_(req, lobby) {
  const doc = normDoc_(req.documento);
  if (!doc) throw error_('Escriba o escanee un número de documento.');
  const v = leer_('Visitantes').filas.find(r => normDoc_(r.documento) === doc) || null;
  const visitas = leer_('Visitas', 20000).filas
    .filter(r => normDoc_(r.documento) === doc)
    .sort((a, b) => fecha_(b.entrada) - fecha_(a.entrada))
    .slice(0, 25)
    .map(visitaPublica_);
  const incidencias = leer_('Incidencias').filas
    .filter(r => normDoc_(r.documento) === doc && s_(r.estatus).toLowerCase() !== 'anulada')
    .sort((a, b) => fecha_(b.fecha) - fecha_(a.fecha))
    .map(r => ({ fecha: r.fecha, torre: s_(r.torre), tipo: s_(r.tipo), gravedad: s_(r.gravedad),
                 descripcion: s_(r.descripcion) }));
  const dentro = dentro_(lobby.torre).find(r => normDoc_(r.documento) === doc) || null;
  return {
    documento: doc,
    visitante: v ? {
      documento: s_(v.documento), tipo_documento: s_(v.tipo_documento), nombre: s_(v.nombre),
      nacionalidad: s_(v.nacionalidad), telefono: s_(v.telefono), empresa: s_(v.empresa),
      primera_visita: v.primera_visita, ultima_visita: v.ultima_visita,
      total_visitas: Number(v.total_visitas) || 0, ultima_categoria: s_(v.ultima_categoria), notas: s_(v.notas),
    } : null,
    visitas: visitas,
    incidencias: incidencias,
    vetado: vetado_(doc, lobby.torre),
    dentro: dentro,
  };
}

/** Un código escaneado: gafete de salida (VST-), invitación (VIS-) o huésped de renta corta (STL-). */
function accionCodigo_(req, lobby) {
  const codigo = s_(req.codigo).toUpperCase().trim();
  if (codigo.indexOf('VST-') === 0) {
    const r = leer_('Visitas', FILAS_RECIENTES).filas.find(x => s_(x.id_visita) === codigo);
    if (!r) throw error_('Ese gafete no está registrado.');
    if (!mismaTorre_(r.torre, lobby.torre)) throw error_('Ese gafete es de otra torre (' + s_(r.torre) + ').');
    return { tipo: s_(r.estatus) === 'dentro' ? 'salida' : 'gafete_cerrado', visita: visitaPublica_(r) };
  }
  if (codigo.indexOf('VIS-') === 0) {
    const inv = leer_('Invitaciones').filas.find(x => s_(x.codigo).toUpperCase() === codigo);
    if (!inv) throw error_('La invitación no existe.');
    if (!mismaTorre_(inv.torre, lobby.torre)) throw error_('La invitación es para ' + s_(inv.torre) + '.');
    const ahora = new Date();
    if (s_(inv.estatus).toLowerCase() === 'anulada') throw error_('La invitación fue anulada.');
    if (inv.valida_desde && fecha_(inv.valida_desde) > ahora) throw error_('La invitación aún no está vigente.');
    if (inv.valida_hasta && fecha_(inv.valida_hasta) < ahora) throw error_('La invitación ya venció.');
    const usosMax = Number(inv.usos_max) || 1;
    if ((Number(inv.usos) || 0) >= usosMax) throw error_('La invitación ya se usó ' + usosMax + ' vez/veces.');
    return { tipo: 'invitacion', datos: {
      codigo: codigo, nombre: s_(inv.visitante), documento: s_(inv.documento), categoria: s_(inv.categoria) || 'VIS',
      destino_apto: s_(inv.apartamento), autorizado_por: 'Invitación ' + codigo, vence: inv.valida_hasta || '',
      notas: s_(inv.notas) } };
  }
  if (codigo.indexOf('STL-') === 0) {
    return { tipo: 'huesped', datos: huespedRentaCorta_(codigo, lobby) };
  }
  throw error_('Código no reconocido.');
}

function accionEntrada_(req, lobby) {
  const v = req.visita || {};
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const recientes = leer_('Visitas', FILAS_RECIENTES).filas;
    const id = /^VST-[A-Z0-9-]{6,30}$/.test(s_(v.id_visita)) ? s_(v.id_visita) : nuevoId_();
    const existente = recientes.find(r => s_(r.id_visita) === id);
    if (existente) return { visita: visitaPublica_(existente), repetida: true };

    const nombre = s_(v.nombre).toUpperCase();
    if (!nombre) throw error_('Falta el nombre del visitante.');
    const cat = leer_('Categorias').filas.find(c => s_(c.codigo) === s_(v.categoria));
    if (!cat) throw error_('Elija el tipo de visitante.');
    if (esSi_(cat.requiere_destino) && !s_(v.destino_apto)) throw error_('Indique el apartamento de destino.');
    if (s_(v.destino_apto)) {
      const apto = leer_('Apartamentos').filas.find(a => mismaTorre_(a.torre, lobby.torre) &&
        s_(a.apartamento).toUpperCase() === s_(v.destino_apto).toUpperCase());
      if (!apto) throw error_('El apartamento ' + v.destino_apto + ' no existe en ' + lobby.torre + '.');
      if (!v.residente) v.residente = s_(apto.residente);
    }
    const doc = s_(v.documento).toUpperCase();
    const veto = vetado_(doc, lobby.torre);
    if (veto) {
      registrar_(lobby, v.portero, 'intento_lista_negra', doc + ' ' + nombre);
      throw error_('ACCESO DENEGADO: la persona está en la lista negra. ' + veto.motivo, 'VETADO');
    }
    const cfgE = config_();
    const individual = !esNo_(cfgE.REGISTRO_INDIVIDUAL);
    if (individual && Number(v.acompanantes) > 0) {
      throw error_('Cada persona que entra debe registrarse con su propio documento. Registre a los acompañantes uno por uno.', 'REGISTRO_INDIVIDUAL');
    }
    if (individual && !doc && !s_(v.codigo_origen)) {
      throw error_('Escanee o escriba el documento de la persona: todo el que entra se registra.', 'REGISTRO_INDIVIDUAL');
    }
    const entrada = new Date();
    const dentroYa = recientes.find(r => doc && normDoc_(r.documento) === normDoc_(doc) &&
      s_(r.estatus) === 'dentro' && mismaTorre_(r.torre, lobby.torre));
    if (dentroYa && !req.forzar) {
      throw error_('Esta persona ya figura adentro desde ' + hora_(dentroYa.entrada) +
        ' (gafete ' + s_(dentroYa.gafete) + '). Registre su salida primero.', 'YA_DENTRO');
    }

    // Vencimiento: límite de horas de la categoría, hora de cierre o vigencia de la invitación
    let vence = Number(cat.horas_max) > 0 ? new Date(entrada.getTime() + Number(cat.horas_max) * 3600000) : null;
    const cierre = cierreDe_(lobby.torre, s_(cat.codigo), entrada);
    if (esSi_(cat.es_personal_trabajo)) {
      if (cierre === 'cerrado' && !req.forzar) {
        throw error_('Hoy no se permite la entrada de ' + s_(cat.nombre).toLowerCase() + ' según el horario.', 'FUERA_HORARIO');
      }
      if (cierre instanceof Date) {
        if (cierre <= entrada && !req.forzar) {
          throw error_('Ya pasó la hora de cierre para ' + s_(cat.nombre).toLowerCase() + ' (' + hora_(cierre) + ').', 'FUERA_HORARIO');
        }
        if (cierre > entrada && (!vence || cierre < vence)) vence = cierre;
      }
    }
    if (v.vence && fecha_(v.vence) > entrada && (!vence || fecha_(v.vence) < vence)) vence = fecha_(v.vence);

    // Ocupación máxima del apartamento de destino
    let capacidad = null;
    if (s_(v.destino_apto) && cuentaOcupacion_(cat)) {
      const oc = ocupacionApto_(lobby.torre, v.destino_apto);
      const n = 1 + (individual ? 0 : Number(v.acompanantes) || 0);
      const total = s_(cat.codigo) === 'HUE'
        ? oc.residentes + oc.visitas + Math.max(oc.hue_visitas + n, oc.reserva_personas)
        : oc.total + n;
      if (oc.max && total > oc.max && !s_(req.autoriza_capacidad)) {
        throw error_('El apartamento ' + s_(v.destino_apto).toUpperCase() + ' ya está en su capacidad máxima (' + oc.total + ' de ' + oc.max +
          ' personas). No registrar sin autorización de la administración.', 'CAPACIDAD');
      }
      capacidad = { apartamento: s_(v.destino_apto).toUpperCase(), total: total, max: oc.max, llena: !!oc.max && total >= oc.max, excedida: !!oc.max && total > oc.max };
    }
    const hoyClave = dia_(entrada);
    const gafete = recientes.filter(r => mismaTorre_(r.torre, lobby.torre) && dia_(r.entrada) === hoyClave).length + 1;
    const fotoUrl = v.foto ? guardarFoto_(v.foto, id) : '';

    const fila = {
      id_visita: id, gafete: ('00' + gafete).slice(-3), torre: lobby.torre, entrada: entrada,
      categoria: s_(cat.codigo), nombre: nombre, tipo_documento: s_(v.tipo_documento), documento: doc,
      empresa: s_(v.empresa), destino_apto: s_(v.destino_apto).toUpperCase(), residente: s_(v.residente),
      autorizado_por: s_(v.autorizado_por), motivo: s_(v.motivo), placa: s_(v.placa).toUpperCase(),
      acompanantes: Number(v.acompanantes) || 0, vence: vence || '', foto_url: fotoUrl,
      codigo_origen: s_(v.codigo_origen), lobby: lobby.nombre, portero: s_(v.portero), salida: '',
      salida_por: '', estatus: 'dentro', notas: s_(v.notas),
    };
    agregar_('Visitas', fila);
    if (doc) actualizarVisitante_(doc, fila, v, fotoUrl);
    if (s_(v.codigo_origen).indexOf('VIS-') === 0) usarInvitacion_(s_(v.codigo_origen));
    registrar_(lobby, v.portero, 'entrada', id + ' ' + nombre + ' → ' + fila.destino_apto);
    if (capacidad && capacidad.excedida) avisarCapacidadExcedida_(lobby, v.portero, capacidad.apartamento, capacidad.total, capacidad.max, s_(req.autoriza_capacidad));
    else if (capacidad && capacidad.llena) registrar_(lobby, v.portero, 'alerta_capacidad', 'Apto ' + capacidad.apartamento + ' en capacidad máxima (' + capacidad.total + '/' + capacidad.max + ')');
    return { visita: visitaPublica_(fila), vetado: null, capacidad: capacidad };
  } finally {
    lock.releaseLock();
  }
}

function accionSalida_(req, lobby) {
  const id = s_(req.id_visita).toUpperCase();
  const t = leer_('Visitas', FILAS_RECIENTES);
  const r = t.filas.find(x => s_(x.id_visita) === id);
  if (!r) throw error_('No encontré esa visita.');
  if (!mismaTorre_(r.torre, lobby.torre)) throw error_('Esa visita es de otra torre.');
  if (s_(r.estatus) !== 'dentro') return { visita: visitaPublica_(r), ya_salio: true };
  const salida = new Date();
  actualizar_('Visitas', r._fila, { salida: salida, salida_por: s_(req.portero), estatus: 'salio' });
  r.salida = salida; r.estatus = 'salio';
  registrar_(lobby, req.portero, 'salida', id + ' ' + s_(r.nombre));
  return { visita: visitaPublica_(r) };
}

function accionIncidencia_(req, lobby) {
  const d = req.incidencia || {};
  if (!s_(d.documento) && !s_(d.nombre)) throw error_('Indique a quién corresponde la incidencia.');
  if (!s_(d.tipo)) throw error_('Elija el tipo de incidencia.');
  const id = 'INC-' + Utilities.formatDate(new Date(), TZ, 'yyMMdd') + '-' + aleatorio_(4);
  agregar_('Incidencias', {
    id_incidencia: id, fecha: new Date(), torre: lobby.torre, documento: s_(d.documento).toUpperCase(),
    nombre: s_(d.nombre).toUpperCase(), id_visita: s_(d.id_visita), tipo: s_(d.tipo),
    gravedad: s_(d.gravedad) || 'leve', descripcion: s_(d.descripcion), reportado_por: s_(d.portero),
    estatus: 'registrada',
  });
  registrar_(lobby, d.portero, 'incidencia', id + ' ' + s_(d.documento) + ' ' + s_(d.tipo));
  let hostAvisado = false;
  try { hostAvisado = reportarAlHost_(id, d, lobby); } catch (err) { hostAvisado = false; }
  const correo = config_().CORREO_ADMINISTRACION;
  if (correo && s_(d.gravedad) === 'grave') {
    try {
      MailApp.sendEmail(correo, 'Incidencia grave en ' + lobby.torre + ' – ' + s_(d.tipo),
        'Visitante: ' + s_(d.nombre) + ' (' + s_(d.documento) + ')\nTipo: ' + s_(d.tipo) +
        '\nDescripción: ' + s_(d.descripcion) + '\nReportado por: ' + s_(d.portero) + '\nLobby: ' + lobby.nombre);
    } catch (err) { /* el registro ya quedó guardado */ }
  }
  return { id_incidencia: id, host_avisado: hostAvisado };
}

function accionInvitacion_(req, lobby) {
  const d = req.invitacion || {};
  if (!s_(d.visitante)) throw error_('Falta el nombre del invitado.');
  if (!s_(d.apartamento)) throw error_('Falta el apartamento.');
  const codigos = leer_('Invitaciones').filas.map(r => s_(r.codigo));
  let codigo;
  do { codigo = 'VIS-' + aleatorio_(6); } while (codigos.indexOf(codigo) >= 0);

/** Hora de cierre para una categoría en una fecha: Date, 'cerrado' o null (sin límite). */
function cierreDe_(torre, categoria, cuando) {
  const sh = hoja_('Horarios');
  if (sh.getLastRow() < 2) return null;
  const vals = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getDisplayValues();
  const enc = vals[0];
  const filas = vals.slice(1).map(v => { const o = {}; enc.forEach((k, j) => o[k] = v[j]); return o; });
  const coincide = (valor, buscado) => !s_(valor) || s_(valor).toUpperCase() === 'TODAS' ||
    s_(valor).toUpperCase() === s_(buscado).toUpperCase();
  const puntaje = r => (s_(r.torre) && s_(r.torre).toUpperCase() !== 'TODAS' ? 2 : 0) +
    (s_(r.categoria) && s_(r.categoria).toUpperCase() !== 'TODAS' ? 1 : 0);
  const candidatas = filas.filter(r => coincide(r.torre, torre) && coincide(r.categoria, categoria))
    .sort((a, b) => puntaje(b) - puntaje(a));
  if (!candidatas.length) return null;
  const dow = Number(Utilities.formatDate(cuando, TZ, 'u')); // 1 = lunes … 7 = domingo
  const valor = s_(dow <= 5 ? candidatas[0].lun_vie : dow === 6 ? candidatas[0].sabado : candidatas[0].domingo).toLowerCase();
  if (!valor) return null;
  if (valor.indexOf('cerr') === 0 || valor === 'no') return 'cerrado';
  const m = valor.match(/(\d{1,2}):(\d{2})\s*(a\.?\s*m\.?|p\.?\s*m\.?)?/);
  if (!m) return null;
  let h = Number(m[1]);
  if (m[3] && m[3].charAt(0) === 'p' && h < 12) h += 12;
  if (m[3] && m[3].charAt(0) === 'a' && h === 12) h = 0;
  const base = Utilities.formatDate(cuando, TZ, 'yyyy-MM-dd');
  return Utilities.parseDate(base + ' ' + ('0' + h).slice(-2) + ':' + m[2], TZ, 'yyyy-MM-dd HH:mm');
}

function cierresHoy_(torre, categorias) {
  const ahora = new Date();
  const out = {};
  categorias.filter(c => c.es_personal_trabajo).forEach(c => {
    const cierre = cierreDe_(torre, c.codigo, ahora);
    out[c.codigo] = cierre === 'cerrado' ? 'cerrado' : cierre ? cierre.toISOString() : null;
  });
  return out;
}

/** Busca un huésped del sistema de check-in de rentas cortas (código STL-). */
function huespedRentaCorta_(codigo, lobby) {
  const id = config_().CHECKIN_SHEET_ID;
  if (!id) return { codigo: codigo, categoria: 'HUE', nombre: '', destino_apto: '',
    aviso: 'Configure CHECKIN_SHEET_ID para traer los datos de la reserva.' };
  const libro = SpreadsheetApp.openById(id);
  const tabla = n => {
    const v = libro.getSheetByName(n).getDataRange().getValues();
    return v.slice(1).map(f => { const o = {}; v[0].forEach((k, j) => o[s_(k)] = f[j]); return o; });
  };
  const r = tabla('Reservas').find(x => s_(x.codigo_reserva).toUpperCase() === codigo);
  if (!r) throw error_('La reserva ' + codigo + ' no existe en el sistema de rentas cortas.');
  if (s_(r.estatus) === 'cancelado') throw error_('La reserva ' + codigo + ' está CANCELADA.');
  const u = tabla('Unidades').find(x => s_(x.codigo_unidad).toUpperCase() === s_(r.codigo_unidad).toUpperCase()) || {};
  if (s_(u.torre) && !mismaTorre_(u.torre, lobby.torre)) throw error_('La reserva es para ' + s_(u.torre) + '.');
  const salida = fecha_(r.salida);
  const hs = s_(config_().HORA_SALIDA_HUESPED || '12:00').split(':');
  if (salida) salida.setHours(Number(hs[0]) || 12, Number(hs[1]) || 0, 0, 0);
  if (salida && salida < new Date()) throw error_('La reserva ' + codigo + ' ya terminó.');
  try {
    reiniciarCache_();
    const ini = inicioEstadia_(r);
    if (ini && ini > new Date()) throw error_('QR todavía no válido: el apartamento recibe desde el ' + Utilities.formatDate(ini, TZ, 'dd/MM/yyyy') +
      ' a las ' + Utilities.formatDate(ini, TZ, 'h:mm a') + '. No dar acceso antes.', 'ANTES_DE_HORA');
  } catch (err) { if (err.codigo === 'ANTES_DE_HORA') throw err; }
  return { codigo: codigo, categoria: 'HUE', nombre: s_(r.huesped_principal), documento: s_(r.documento_principal),
    destino_apto: s_(u.apartamento), autorizado_por: 'Reserva ' + codigo, vence: salida || '' };
}

function instalar() {
  const libro = libro_();
  Object.keys(HOJAS).forEach(n => {
    let sh = libro.getSheetByName(n);
    if (!sh) {
      sh = libro.insertSheet(n);
      sh.getRange(1, 1, 1, HOJAS[n].length).setValues([HOJAS[n]]).setFontWeight('bold');
      sh.setFrozenRows(1);
    } else {
      const enc = sh.getRange(1, 1, 1, Math.max(sh.getLastColumn(), 1)).getValues()[0].map(String);
      const faltan = HOJAS[n].filter(c => enc.indexOf(c) < 0);
      if (faltan.length) sh.getRange(1, sh.getLastColumn() + 1, 1, faltan.length).setValues([faltan]).setFontWeight('bold');
    }
  });
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'archivar')
    .forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('archivar').timeBased().onMonthDay(1).atHour(3).create();
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'sincronizar').forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('sincronizar').timeBased().everyHours(1).create();
  carpetaFotos_();
  if (!leer_('Admins').filas.length) {
    agregar_('Admins', { nombre: 'Gerencia', correo: Session.getActiveUser().getEmail(), token: 'ADM-' + aleatorio_(12), rol: 'gerente', activo: 'SI' });
  }
  const ci = libroCheckin_();
  if (ci) {
    Object.keys(TABLAS_CHECKIN_NUEVAS).forEach(n => {
      if (!ci.getSheetByName(n)) {
        const sh = ci.insertSheet(n);
        sh.getRange(1, 1, 1, TABLAS_CHECKIN_NUEVAS[n].length).setValues([TABLAS_CHECKIN_NUEVAS[n]]).setFontWeight('bold');
        sh.setFrozenRows(1);
      }
    });
    Object.keys(COLUMNAS_CHECKIN).forEach(n => {
      const sh = ci.getSheetByName(n); if (!sh) return;
      const enc = sh.getRange(1, 1, 1, Math.max(sh.getLastColumn(), 1)).getValues()[0].map(String);
      const faltan = COLUMNAS_CHECKIN[n].filter(c => enc.indexOf(c) < 0);
      if (faltan.length) sh.getRange(1, sh.getLastColumn() + 1, 1, faltan.length).setValues([faltan]).setFontWeight('bold');
    });
  }
  try { sincronizar(); } catch (err) { Logger.log('Sincronización con el Check-in pendiente: ' + err.message); }
  fijarConfig_('VERSION_SERVICIO', VERSION_SERVICIO);
  Logger.log('Listo. Versión ' + VERSION_SERVICIO + '. Pestañas revisadas y archivo mensual programado. Token de administración en la pestaña Admins.');
}

/** Mueve a un archivo aparte las visitas cerradas con más de DIAS_EN_HOJA días (60 por defecto). */
function archivar() {
  const dias = Number(config_().DIAS_EN_HOJA) || 60;
  const corte = new Date(Date.now() - dias * 86400000);
  const sh = hoja_('Visitas');
  const ultima = sh.getLastRow();
  if (ultima < 2) return;
  const enc = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String);
  const datos = sh.getRange(2, 1, ultima - 1, enc.length).getValues();
  const iEnt = enc.indexOf('entrada'), iEst = enc.indexOf('estatus');
  let n = 0;
  while (n < datos.length && fecha_(datos[n][iEnt]) < corte && s_(datos[n][iEst]) !== 'dentro') n++;
  if (!n) return;
  const nombre = 'Visitas – Archivo ' + Utilities.formatDate(new Date(), TZ, 'yyyy-MM');
  const padre = DriveApp.getFileById(libro_().getId()).getParents();
  const destino = SpreadsheetApp.create(nombre);
  const hojaDestino = destino.getSheets()[0];
  hojaDestino.getRange(1, 1, 1, enc.length).setValues([enc]);
  hojaDestino.getRange(2, 1, n, enc.length).setValues(datos.slice(0, n));
  if (padre.hasNext()) DriveApp.getFileById(destino.getId()).moveTo(padre.next());
  sh.deleteRows(2, n);
  registrar_({ nombre: 'sistema' }, '', 'archivo', n + ' visitas movidas a ' + nombre);
}

// Utilidades de la hoja
// ===========================================================================
function libro_() { return SpreadsheetApp.getActiveSpreadsheet(); }

function hoja_(nombre) {
  const sh = libro_().getSheetByName(nombre);
  if (!sh) throw error_('Falta la pestaña «' + nombre + '». Ejecute instalar() en Apps Script.');
  return sh;
}

function encabezados_(sh) {
  return sh.getRange(1, 1, 1, Math.max(sh.getLastColumn(), 1)).getValues()[0].map(x => s_(x));
}

function leer_(nombre, ultimas) {
  const sh = hoja_(nombre);
  const enc = encabezados_(sh);
  const last = sh.getLastRow();
  if (last < 2) return { sh: sh, enc: enc, filas: [] };
  const inicio = ultimas && last - 1 > ultimas ? last - ultimas + 1 : 2;
  const vals = sh.getRange(inicio, 1, last - inicio + 1, enc.length).getValues();
  const filas = [];
  vals.forEach((v, i) => {
    if (!v.some(x => x !== '' && x !== null)) return;
    const o = { _fila: inicio + i };
    enc.forEach((k, j) => { if (k) o[k] = v[j]; });
    filas.push(o);
  });
  return { sh: sh, enc: enc, filas: filas };
}

function agregar_(nombre, obj) {
  const sh = hoja_(nombre);
  sh.appendRow(encabezados_(sh).map(k => celda_(obj[k], k)));
}

function actualizar_(nombre, fila, cambios) {
  const sh = hoja_(nombre);
  const enc = encabezados_(sh);
  Object.keys(cambios).forEach(k => {
    const c = enc.indexOf(k);
    if (c >= 0) sh.getRange(fila, c + 1).setValue(celda_(cambios[k], k));
  });
}

// Columnas de texto largo (el resto se corta a 500 caracteres)
const LARGO_COLUMNA = { resumen: 45000, texto: 4000, normas: 2000, notas_saliente: 2000, instrucciones: 1000, descripcion: 1000, autorizados: 1000 };
function celda_(v, columna) {
  if (v === undefined || v === null) return '';
  if (v instanceof Date || typeof v === 'number' || typeof v === 'boolean') return v;
  const t = String(v).slice(0, LARGO_COLUMNA[columna] || 500);
  // Evita fórmulas inyectadas y que documentos o teléfonos pierdan ceros
  if (/^[=+\-@]/.test(t) || /^\d[\d\- ]*$/.test(t)) return "'" + t;
  return t;
}

function config_() {
  const out = {};
  leer_('Config').filas.forEach(r => { if (s_(r.clave)) out[s_(r.clave).toUpperCase()] = s_(r.valor); });
  return out;
}

function fijarConfig_(clave, valor) {
  const r = leer_('Config').filas.find(x => s_(x.clave).toUpperCase() === clave);
  if (r) actualizar_('Config', r._fila, { valor: valor });
  else agregar_('Config', { clave: clave, valor: valor, descripcion: 'Creado automáticamente' });
}

function registrar_(lobby, portero, accion, detalle) {
  try {
    agregar_('Bitacora', { fecha_hora: new Date(), lobby: lobby.nombre || '', portero: s_(portero),
      accion: accion, detalle: s_(detalle) });
  } catch (err) { /* la bitácora nunca detiene una operación */ }
}

function s_(v) {
  if (v === undefined || v === null) return '';
  if (v instanceof Date) return v.toISOString();
  return String(v).trim();
}

function esSi_(v) { return ['SI', 'SÍ', 'S', 'TRUE', 'VERDADERO', '1', 'X'].indexOf(s_(v).toUpperCase()) >= 0; }
function esNo_(v) { return ['NO', 'N', 'FALSE', 'FALSO', '0'].indexOf(s_(v).toUpperCase()) >= 0; }

function mismaTorre_(a, b) { return s_(a).toUpperCase() === s_(b).toUpperCase(); }

function normDoc_(v) {
  const x = s_(v).toUpperCase().replace(/[^A-Z0-9]/g, '');
  return /^\d+$/.test(x) ? x.replace(/^0+/, '') : x;
}

function fecha_(v) {
  if (!v) return null;
  if (v instanceof Date) return new Date(v.getTime());
  const d = new Date(v);
  return isNaN(d.getTime()) ? null : d;
}

function dia_(v) {
  const d = fecha_(v);
  return d ? Utilities.formatDate(d, TZ, 'yyyy-MM-dd') : '';
}

function hora_(v) {
  const d = fecha_(v);
  return d ? Utilities.formatDate(d, TZ, 'h:mm a') : '';
}

function aleatorio_(n) {
  const alfabeto = 'ABCDEFGHJKMNPQRSTUVWXYZ23456789';
  let t = '';
  for (let i = 0; i < n; i++) t += alfabeto.charAt(Math.floor(Math.random() * alfabeto.length));
  return t;
}

function nuevoId_() {
  return 'VST-' + Utilities.formatDate(new Date(), TZ, 'yyMMdd') + '-' + aleatorio_(5);
}

let _ci = null;
function libroCheckin_() {
  if (_ci !== null) return _ci || null;
  const id = config_().CHECKIN_SHEET_ID;
  try { _ci = id ? SpreadsheetApp.openById(id) : false; } catch (err) { _ci = false; }

const _cacheCI = {};
function reiniciarCache_() { Object.keys(_cacheCI).forEach(k => delete _cacheCI[k]); _ci = null; }
function leerCI_(nombre) {
  if (_cacheCI[nombre]) return _cacheCI[nombre];
  const r = leerCISinCache_(nombre);
  _cacheCI[nombre] = r;
  return r;
}
function leerCISinCache_(nombre) {
  const libro = libroCheckin_();
  const sh = libro && libro.getSheetByName(nombre);
  if (!sh || sh.getLastRow() < 1) return { sh: sh, enc: [], filas: [] };
  const vals = sh.getDataRange().getValues();
  const enc = vals[0].map(x => s_(x));
  const filas = [];
  vals.slice(1).forEach((v, i) => {
    if (!v.some(x => x !== '' && x !== null)) return;
    const o = { _fila: i + 2 };
    enc.forEach((k, j) => { if (k) o[k] = v[j]; });
    filas.push(o);
  });
  return { sh: sh, enc: enc, filas: filas };
}
function agregarCI_(nombre, obj) {
  const t = leerCI_(nombre);
  if (!t.sh) throw error_('Falta la pestaña «' + nombre + '» en la hoja del Check-in.');
  t.sh.appendRow(t.enc.map(k => celda_(obj[k], k)));
  delete _cacheCI[nombre];
}
function actualizarCI_(nombre, fila, cambios) {
  const t = leerCI_(nombre);
  Object.keys(cambios).forEach(k => {
    const c = t.enc.indexOf(k);
    if (c >= 0) t.sh.getRange(fila, c + 1).setValue(celda_(cambios[k], k));
  });
  delete _cacheCI[nombre];
}
function normTel_(v) { return s_(v).replace(/\D/g, '').slice(-10); }
function torreDeUnidad_(unidades, codigo) {
  const u = unidades.find(x => s_(x.codigo_unidad).toUpperCase() === s_(codigo).toUpperCase());
  return u ? { torre: s_(u.torre), apto: s_(u.apartamento), activa: s_(u.estatus_unidad || 'activa').toLowerCase() !== 'inactiva' } : null;
}
function horaSalidaCI_() {
  const c = leerCI_('Config').filas.find(r => s_(r.clave).toUpperCase() === 'HORA_SALIDA');
  const v = c ? c.valor : '12:00';
  if (v instanceof Date) return [v.getHours(), v.getMinutes()];
  if (typeof v === 'number') { const m = Math.round(v * 1440); return [Math.floor(m / 60) % 24, m % 60]; }
  const m = s_(v).match(/(\d{1,2}):(\d{2})/);
  return m ? [Number(m[1]), Number(m[2])] : [12, 0];
}
function finEstadia_(reserva, hs) {

  return Utilities.parseDate(f + ' ' + ('0' + hs[0]).slice(-2) + ':' + ('0' + hs[1]).slice(-2), TZ, 'yyyy-MM-dd HH:mm');
}

// ---------------------------------------------------------------------------
// Reglas contra conflictos (las mismas que aplica el bot)
//   R1 Un teléfono pertenece a un solo host.
//   R2 Una cuenta de Telegram pertenece a un solo host.
//   R3 Un apartamento tiene a lo sumo UN host activo. Si tiene dos, queda bloqueado.
//   R4 Un permiso debe apuntar a una unidad y a un host que existan.
//   R5 El código de unidad es único; cada apartamento existe una sola vez por torre.
// ---------------------------------------------------------------------------
function auditarConflictos_() {
  const out = [];
  const hosts = leerCI_('Hosts').filas, permisos = leerCI_('Permisos').filas, unidades = leerCI_('Unidades').filas;
  const activos = hosts.filter(h => s_(h.estatus).toLowerCase() !== 'suspendido');
  const agrupar = (lista, clave) => { const m = {}; lista.forEach(x => { const k = clave(x); if (k) (m[k] = m[k] || []).push(x); }); return m; };
  const porTel = agrupar(activos, h => normTel_(h.telefono));
  Object.keys(porTel).forEach(k => { if (porTel[k].length > 1) out.push({ regla: 'R1', tipo: 'Teléfono repetido', gravedad: 'alta',
    detalle: 'El teléfono …' + k.slice(-4) + ' está en ' + porTel[k].length + ' hosts: ' + porTel[k].map(h => s_(h.id_host)).join(', '),
    ids: porTel[k].map(h => s_(h.id_host)) }); });
  const porTg = agrupar(activos, h => s_(h.telegram_id));
  Object.keys(porTg).forEach(k => { if (porTg[k].length > 1) out.push({ regla: 'R2', tipo: 'Cuenta de Telegram repetida', gravedad: 'alta',
    detalle: 'La misma cuenta de Telegram está en ' + porTg[k].map(h => s_(h.id_host)).join(', '), ids: porTg[k].map(h => s_(h.id_host)) }); });
  const idsHost = {}; hosts.forEach(h => { idsHost[s_(h.id_host)] = h; });
  const codUnidad = {}; unidades.forEach(u => { const c = s_(u.codigo_unidad).toUpperCase(); codUnidad[c] = (codUnidad[c] || 0) + 1; });
  Object.keys(codUnidad).forEach(c => { if (c && codUnidad[c] > 1) out.push({ regla: 'R5', tipo: 'Código de unidad repetido', gravedad: 'alta',
    detalle: 'El código ' + c + ' aparece ' + codUnidad[c] + ' veces en Unidades', unidad: c }); });
  const act = permisos.filter(p => esSi_(p.activo));
  const porUnidad = agrupar(act, p => s_(p.codigo_unidad).toUpperCase());
  Object.keys(porUnidad).forEach(c => {
    const distintos = Array.from(new Set(porUnidad[c].map(p => s_(p.id_host))));
    if (distintos.length > 1) out.push({ regla: 'R3', tipo: 'Apartamento con dos hosts', gravedad: 'alta',
      detalle: c + ' tiene acceso activo de ' + distintos.join(' y ') + '. La unidad queda bloqueada en el bot hasta resolverlo.', unidad: c, ids: distintos });
    if (!codUnidad[c]) out.push({ regla: 'R4', tipo: 'Permiso a unidad inexistente', gravedad: 'media', detalle: 'Hay permiso activo para ' + c + ', que no está en Unidades', unidad: c });
    porUnidad[c].forEach(p => { if (!idsHost[s_(p.id_host)]) out.push({ regla: 'R4', tipo: 'Permiso de host inexistente', gravedad: 'media',
      detalle: 'El permiso de ' + c + ' apunta al host ' + s_(p.id_host) + ', que no existe', unidad: c }); });
  });
  const aptos = agrupar(leer_('Apartamentos').filas, a => s_(a.torre).toUpperCase() + '|' + s_(a.apartamento).toUpperCase());
  Object.keys(aptos).forEach(k => { if (aptos[k].length > 1) out.push({ regla: 'R5', tipo: 'Apartamento repetido', gravedad: 'media',
    detalle: k.replace('|', ' · apto ') + ' aparece ' + aptos[k].length + ' veces en Apartamentos' }); });
  return out;
}

/** ¿Puede este teléfono quedarse con esta unidad? Devuelve null si sí, o el motivo si no. */
function conflictoAsignacion_(codigoUnidad, telefono) {
  const hosts = leerCI_('Hosts').filas.filter(h => s_(h.estatus).toLowerCase() !== 'suspendido');
  const mismos = hosts.filter(h => normTel_(h.telefono) === normTel_(telefono));
  if (mismos.length > 1) return { tipo: 'TEL_DUPLICADO', texto: 'Ese teléfono ya está repetido en varios hosts.' };
  const idHost = mismos.length ? s_(mismos[0].id_host) : '';
  const otros = leerCI_('Permisos').filas.filter(p => esSi_(p.activo) && s_(p.codigo_unidad).toUpperCase() === s_(codigoUnidad).toUpperCase() && s_(p.id_host) !== idHost);
  if (otros.length) {
    const h = hosts.find(x => s_(x.id_host) === s_(otros[0].id_host));
    return { tipo: 'UNIDAD_CON_OTRO_HOST', texto: 'La unidad ya la administra ' + (h ? s_(h.nombre) + ' (' + s_(h.id_host) + ')' : s_(otros[0].id_host)) + '.', id_host_actual: s_(otros[0].id_host) };
  }
  return null;
}

function nuevoIdHost_(hosts) {
  let n = 1;
  hosts.forEach(h => { const m = s_(h.id_host).match(/(\d+)/); if (m) n = Math.max(n, Number(m[1]) + 1); });
  return 'H' + ('00' + n).slice(-3);
}

/** Asigna una unidad a un host, desactivando a cualquier otro. Todo dentro de un bloqueo. */
function asignarUnidad_(codigo, idHost, quien, transferir) {
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const permisos = leerCI_('Permisos').filas;
    const activos = permisos.filter(p => esSi_(p.activo) && s_(p.codigo_unidad).toUpperCase() === codigo.toUpperCase());
    const ajenos = activos.filter(p => s_(p.id_host) !== idHost);
    if (ajenos.length && !transferir) throw error_('La unidad ' + codigo + ' ya tiene otro host activo. Use «Transferir».', 'CONFLICTO');
    ajenos.forEach(p => actualizarCI_('Permisos', p._fila, { activo: 'NO' }));
    if (!activos.some(p => s_(p.id_host) === idHost)) {
      const previo = permisos.find(p => s_(p.id_host) === idHost && s_(p.codigo_unidad).toUpperCase() === codigo.toUpperCase());
      if (previo) actualizarCI_('Permisos', previo._fila, { activo: 'SI', fecha_autorizacion: new Date(), autorizado_por: quien });
      else agregarCI_('Permisos', { id_host: idHost, codigo_unidad: codigo.toUpperCase(), activo: 'SI', fecha_autorizacion: new Date(), autorizado_por: quien });
    }
    registrar_({ nombre: 'admin' }, quien, transferir && ajenos.length ? 'transferencia' : 'asignacion',
      codigo + ' → ' + idHost + (ajenos.length ? ' (antes ' + ajenos.map(p => s_(p.id_host)).join(', ') + ')' : ''));
    return { desactivados: ajenos.map(p => s_(p.id_host)) };
  } finally { lock.releaseLock(); }
}

function accionTransferir_(req, admin) {
  const codigo = s_(req.codigo_unidad).toUpperCase(), idHost = s_(req.id_host);
  if (!codigo || !idHost) throw error_('Indique la unidad y el host nuevo.');
  if (!leerCI_('Hosts').filas.some(h => s_(h.id_host) === idHost)) throw error_('El host ' + idHost + ' no existe.');
  return asignarUnidad_(codigo, idHost, admin.nombre, true);
}

const TIPOS_HOST = ['host', 'propietario'];
function vistaHost_(h, permisos, unidades, gerente) {
  const mias = permisos.filter(p => s_(p.id_host) === s_(h.id_host)).map(p => {
    const c = s_(p.codigo_unidad).toUpperCase(), u = torreDeUnidad_(unidades, c) || {};
    return { codigo: c, torre: u.torre || '', apto: u.apto || '' };
  });
  return { id_host: s_(h.id_host), nombre: s_(h.nombre), tipo: TIPOS_HOST.indexOf(s_(h.tipo).toLowerCase()) >= 0 ? s_(h.tipo).toLowerCase() : 'host',
    telefono: gerente ? s_(h.telefono) : enmascararDoc_(h.telefono), correo: gerente ? s_(h.correo) : '',
    documento: gerente ? s_(h.documento) : enmascararDoc_(h.documento), estatus: s_(h.estatus).toLowerCase() || 'pendiente',
    vinculado: !!s_(h.telegram_id), fecha_registro: h.fecha_registro || '', notas: gerente ? s_(h.notas) : '', unidades: mias };
}
function accionHosts_(req, admin) {
  if (!libroCheckin_()) throw error_('Configure CHECKIN_SHEET_ID para administrar hosts.');
  const g = admin.rol === 'gerente';
  const permisos = leerCI_('Permisos').filas.filter(p => esSi_(p.activo)), unidades = leerCI_('Unidades').filas;
  const hosts = leerCI_('Hosts').filas.map(h => vistaHost_(h, permisos, unidades, g))
    .sort((a, b) => (a.estatus === 'suspendido') - (b.estatus === 'suspendido') || a.nombre.localeCompare(b.nombre));
  return { hosts: hosts, gerente: g, bot: s_(configCI_('BOT_USUARIO')).replace(/^@/, ''), empresa: configPublica_().NOMBRE_EMPRESA };
}
function accionHostGuardar_(req, admin) {
  if (!libroCheckin_()) throw error_('Configure CHECKIN_SHEET_ID para administrar hosts.');
  const d = req.host || {};
  const nombre = s_(d.nombre).replace(/\s+/g, ' ').slice(0, 80), tel = s_(d.telefono).replace(/[^\d+]/g, '');
  if (nombre.length < 3) throw error_('Escriba el nombre completo.');
  if (normTel_(tel).length < 10) throw error_('Escriba el teléfono con código de área (10 dígitos o más). Debe ser el mismo de su Telegram.');
  if (s_(d.correo) && !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(s_(d.correo))) throw error_('El correo no es válido.');
  const tipo = TIPOS_HOST.indexOf(s_(d.tipo).toLowerCase()) >= 0 ? s_(d.tipo).toLowerCase() : 'host';
  const estatus = s_(d.estatus).toLowerCase() === 'suspendido' ? 'suspendido' : 'aprobado';
  const doc = s_(d.documento).toUpperCase().slice(0, 30);
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const hosts = leerCI_('Hosts').filas;
    const actual = s_(d.id_host) ? hosts.find(h => s_(h.id_host) === s_(d.id_host)) : null;
    if (s_(d.id_host) && !actual) throw error_('El host ' + s_(d.id_host) + ' no existe.');
    if (estatus !== 'suspendido') {
      const otro = hosts.find(h => h !== actual && normTel_(h.telefono) === normTel_(tel) && s_(h.estatus).toLowerCase() !== 'suspendido');
      if (otro) throw error_('Ese teléfono ya pertenece a ' + s_(otro.nombre) + ' (' + s_(otro.id_host) + '). Un teléfono = un host (R1).', 'CONFLICTO');
      const veto = doc && vetado_(doc, '');
      if (veto) throw error_('ACCESO DENEGADO: ese documento está en la lista negra (' + veto.motivo + ').', 'VETADO');
    }
    const datos = { nombre: nombre, telefono: tel, correo: s_(d.correo).slice(0, 100), documento: doc, tipo: tipo, estatus: estatus, notas: s_(d.notas).slice(0, 300) };
    let id;
    if (actual) {
      id = s_(actual.id_host);
      // Si cambia el teléfono, la cuenta de Telegram anterior deja de valer: el host debe volver a compartir su número.
      if (normTel_(actual.telefono) !== normTel_(tel)) { datos.telegram_id = ''; datos.token_panel = ''; }
      if (estatus === 'suspendido') datos.token_panel = '';
      actualizarCI_('Hosts', actual._fila, datos);
    } else {
      id = nuevoIdHost_(hosts);
      agregarCI_('Hosts', Object.assign({ id_host: id, fecha_registro: new Date(), telegram_id: '', token_panel: '' }, datos));
    }
    registrar_({ nombre: 'admin' }, admin.nombre, actual ? 'host_editado' : 'host_creado', id + ' ' + nombre + ' (' + tipo + ', ' + estatus + ')');
    const permisos = leerCI_('Permisos').filas.filter(p => esSi_(p.activo));
    return { host: vistaHost_(leerCI_('Hosts').filas.find(h => s_(h.id_host) === id), permisos, leerCI_('Unidades').filas, true), nuevo: !actual };
  } finally { lock.releaseLock(); }
}
function accionHostUnidad_(req, admin) {
  if (!libroCheckin_()) throw error_('Configure CHECKIN_SHEET_ID para administrar hosts.');
  const idHost = s_(req.id_host);
  const h = leerCI_('Hosts').filas.find(x => s_(x.id_host) === idHost);
  if (!h) throw error_('El host ' + idHost + ' no existe.');
  if (req.operacion === 'quitar') {
    const codigo = s_(req.codigo_unidad).toUpperCase();
    const ps = leerCI_('Permisos').filas.filter(p => s_(p.id_host) === idHost && s_(p.codigo_unidad).toUpperCase() === codigo && esSi_(p.activo));
    if (!ps.length) throw error_('Ese apartamento no está asignado a ' + s_(h.nombre) + '.');
    ps.forEach(p => actualizarCI_('Permisos', p._fila, { activo: 'NO' }));
    registrar_({ nombre: 'admin' }, admin.nombre, 'permiso_retirado', codigo + ' ← ' + idHost);
    return { quitado: codigo };
  }
  if (s_(h.estatus).toLowerCase() === 'suspendido') throw error_('El host está suspendido. Reactívelo antes de asignarle apartamentos.');
  const torre = s_(req.torre), apto = s_(req.apartamento).toUpperCase();
  if (!torre || !apto) throw error_('Elija la torre y el apartamento.');
  const ap = leer_('Apartamentos').filas.find(a => mismaTorre_(a.torre, torre) && s_(a.apartamento).toUpperCase() === apto);
  if (!ap) throw error_('El apartamento ' + apto + ' no existe en ' + torre + '. Cárguelo primero en «Cargar torre».');
  const unidades = leerCI_('Unidades').filas;
  let u = unidades.find(x => mismaTorre_(x.torre, torre) && s_(x.apartamento).toUpperCase() === apto);
  const codigo = u ? s_(u.codigo_unidad).toUpperCase() : prefijoTorre_(torre) + '-' + apto;
  if (!u) agregarCI_('Unidades', { codigo_unidad: codigo, torre: s_(ap.torre), apartamento: apto, estatus_unidad: 'activa',
    ocupacion_maxima: Number(ap.ocupacion_maxima) || '', notas: 'Alta desde el Panel · ' + admin.nombre });
  const asig = asignarUnidad_(codigo, idHost, admin.nombre, !!req.transferir);
  return { codigo_unidad: codigo, desactivados: asig.desactivados };
}

// Cargar una torre: genera apartamentos, código de encuesta y código de lobby

function generarAptos_(req) {
  if (s_(req.lista)) return s_(req.lista).split(/[\s,;]+/).map(x => x.trim().toUpperCase()).filter(Boolean);
  const desde = Number(req.piso_desde) || 1, hasta = Number(req.piso_hasta) || 0;
  const letras = s_(req.letras || 'A,B,C,D').split(/[\s,;]+/).filter(Boolean).map(x => x.toUpperCase());
  const out = [];
  for (let p = desde; p <= hasta; p++) letras.forEach(l => out.push(p + l));
  return out;
}
function accionCargarTorre_(req, admin) {
  const torre = s_(req.torre);
  if (!torre) throw error_('Escriba el nombre de la torre.');
  const prefijo = (s_(req.prefijo) || torre.replace(/^torre\s+/i, '')).toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 10);
  const aptos = Array.from(new Set(generarAptos_(req)));
  if (!aptos.length) throw error_('No hay apartamentos para cargar. Indique pisos y letras, o pegue la lista.');
  if (aptos.length > 600) throw error_('Máximo 600 apartamentos por carga.');
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const existentes = {}; leer_('Apartamentos').filas.filter(a => mismaTorre_(a.torre, torre)).forEach(a => { existentes[s_(a.apartamento).toUpperCase()] = true; });
    const nuevos = aptos.filter(a => !existentes[a]);
    if (nuevos.length) {
      const sh = hoja_('Apartamentos'), enc = encabezados_(sh);
      const capDef = Math.max(0, Math.min(50, Number(req.ocupacion_maxima) || 0)) || '';
      const filas = nuevos.map(a => enc.map(k => celda_({ torre: torre, apartamento: a, activo: 'SI', ocupacion_maxima: capDef }[k])));
      sh.getRange(sh.getLastRow() + 1, 1, filas.length, enc.length).setValues(filas);
    }
    let t = leer_('Torres').filas.find(r => mismaTorre_(r.torre, torre));
    let codigoEncuesta = t ? s_(t.codigo_encuesta) : '';
    if (!t) {
      codigoEncuesta = prefijo.slice(0, 6) + '-' + aleatorio_(6);
      agregar_('Torres', { torre: torre, prefijo: prefijo, codigo_encuesta: codigoEncuesta, encuesta_abierta: 'SI',
        pisos: s_(req.piso_desde) && s_(req.piso_hasta) ? req.piso_desde + '-' + req.piso_hasta : '', aptos_por_piso: s_(req.letras), activa: 'SI' });
    }
    let lobby = leer_('Lobbies').filas.find(r => mismaTorre_(r.torre, torre));
    const codigoLobby = lobby ? s_(lobby.codigo_lobby) : prefijo.slice(0, 6) + '-' + aleatorio_(6);
    if (!lobby) agregar_('Lobbies', { codigo_lobby: codigoLobby, torre: torre, nombre_lobby: 'Lobby ' + torre, activo: 'SI' });
    registrar_({ nombre: 'admin' }, admin.nombre, 'cargar_torre', torre + ': ' + nuevos.length + ' apartamentos nuevos');
    return { torre: torre, prefijo: prefijo, apartamentos_nuevos: nuevos.length, ya_existian: aptos.length - nuevos.length,
      codigo_encuesta: codigoEncuesta, codigo_lobby: codigoLobby };
  } finally { lock.releaseLock(); }
}

// Encuesta de cada torre (pública con el código de la torre)

function torrePorCodigoEncuesta_(codigo) {
  const t = leer_('Torres').filas.find(r => s_(r.codigo_encuesta).toUpperCase() === s_(codigo).toUpperCase());
  if (!t) throw error_('El enlace de la encuesta no es válido. Pida uno nuevo a la administración.', 'SIN_ENCUESTA');
  if (!esSi_(t.encuesta_abierta)) throw error_('La encuesta de esta torre está cerrada.', 'SIN_ENCUESTA');
  return t;
}

function accionEncuestaEnviar_(req) {
  const t = torrePorCodigoEncuesta_(req.codigo);
  const d = req.respuesta || {};
  const apto = s_(d.apartamento).toUpperCase();
  const existe = leer_('Apartamentos').filas.some(a => mismaTorre_(a.torre, t.torre) && s_(a.apartamento).toUpperCase() === apto);
  if (!existe) throw error_('El apartamento ' + apto + ' no está en la lista de ' + s_(t.torre) + '.');
  if (!s_(d.propietario) || normTel_(d.telefono).length < 10) throw error_('Escriba el nombre del propietario y un teléfono de 10 dígitos.');
  if (!esSi_(d.acepta_privacidad)) throw error_('Debe aceptar el aviso de privacidad para enviar la encuesta.');
  const renta = esSi_(d.es_renta_corta);
  const hostProp = esSi_(d.host_es_propietario);
  if (renta && !hostProp && (!s_(d.host_nombre) || normTel_(d.host_telefono).length < 10)) throw error_('Indique el nombre y el teléfono de quien administra la renta corta.');
  // Límite: 5 envíos por apartamento por hora
  const cache = CacheService.getScriptCache(), k = 'enc|' + s_(t.torre) + '|' + apto, n = Number(cache.get(k) || 0);
  if (n >= 5) throw error_('Demasiados envíos para este apartamento. Intente más tarde.');
  cache.put(k, String(n + 1), 3600);
  const fila = {
    id: 'ENC-' + Utilities.formatDate(new Date(), TZ, 'yyMMdd') + '-' + aleatorio_(4), fecha: new Date(), torre: s_(t.torre), apartamento: apto,
    propietario: s_(d.propietario).toUpperCase(), telefono: s_(d.telefono), correo: s_(d.correo), vive_en_apto: esSi_(d.vive_en_apto) ? 'SI' : 'NO',
    personas_viven: Math.max(0, Math.min(30, Number(d.personas_viven) || 0)) || '',
    es_renta_corta: renta ? 'SI' : 'NO', host_es_propietario: renta ? (hostProp ? 'SI' : 'NO') : '',
    host_nombre: renta ? s_(hostProp ? d.propietario : d.host_nombre).toUpperCase() : '', host_telefono: renta ? s_(hostProp ? d.telefono : d.host_telefono) : '',
    host_correo: renta ? s_(hostProp ? d.correo : d.host_correo) : '', autorizados: s_(d.autorizados).slice(0, 500), instrucciones: s_(d.instrucciones).slice(0, 300),
    acepta_privacidad: 'SI', estatus: 'pendiente', conflicto: '',
  };
  const previas = leer_('Encuestas').filas.filter(r => mismaTorre_(r.torre, t.torre) && s_(r.apartamento).toUpperCase() === apto && s_(r.estatus) === 'pendiente');
  if (previas.length) fila.conflicto = 'Hay ' + previas.length + ' respuesta(s) pendiente(s) para este apartamento.';
  agregar_('Encuestas', fila);
  return { id: fila.id };
}

function evaluarEncuesta_(r) {
  const avisos = [];
  if (esSi_(r.es_renta_corta)) {
    const prefijo = prefijoTorre_(r.torre);
    const c = conflictoAsignacion_(prefijo + '-' + s_(r.apartamento).toUpperCase(), r.host_telefono);
    if (c) avisos.push(c.texto);
  }
  const otras = leer_('Encuestas').filas.filter(x => x.id !== r.id && mismaTorre_(x.torre, r.torre) && s_(x.apartamento) === s_(r.apartamento) && ['pendiente', 'conflicto'].indexOf(s_(x.estatus)) >= 0);
  if (otras.length) avisos.push('Otra respuesta pendiente para el mismo apartamento (' + otras.map(x => s_(x.propietario)).join(', ') + ').');
  return avisos;
}
function prefijoTorre_(torre) {
  const t = leer_('Torres').filas.find(r => mismaTorre_(r.torre, torre));
  return (t && s_(t.prefijo)) || s_(torre).replace(/^torre\s+/i, '').toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 10);
}
function accionListarEncuestas_(req) {
  const est = s_(req.estatus || 'pendiente');
  const filas = leer_('Encuestas').filas.filter(r => est === 'todas' || s_(r.estatus) === est || (est === 'pendiente' && s_(r.estatus) === 'conflicto'))
    .sort((a, b) => fecha_(b.fecha) - fecha_(a.fecha)).slice(0, 200);
  return { encuestas: filas.map(r => {
    const o = {}; HOJAS.Encuestas.forEach(k => { o[k] = r[k] instanceof Date ? r[k].toISOString() : s_(r[k]); });
    o.avisos = ['pendiente', 'conflicto'].indexOf(o.estatus) >= 0 ? evaluarEncuesta_(r) : [];
    return o;
  }) };
}
function accionDecidirEncuesta_(req, admin) {
  const r = leer_('Encuestas').filas.find(x => s_(x.id) === s_(req.id));
  if (!r) throw error_('No encontré esa respuesta.');
  if (['aprobada', 'rechazada'].indexOf(s_(r.estatus)) >= 0) throw error_('Esa respuesta ya fue ' + s_(r.estatus) + '.');
  if (req.decision === 'rechazar') {
    actualizar_('Encuestas', r._fila, { estatus: 'rechazada', revisado_por: admin.nombre, fecha_revision: new Date() });
    return { estatus: 'rechazada' };
  }
  // 1) Directorio de residentes (Visitantes)
  const ap = leer_('Apartamentos').filas.find(a => mismaTorre_(a.torre, r.torre) && s_(a.apartamento).toUpperCase() === s_(r.apartamento).toUpperCase());
  const cambiosApto = { residente: s_(r.propietario), telefono: s_(r.telefono) };
  if (Number(r.personas_viven) > 0) cambiosApto.residentes_habituales = Number(r.personas_viven);
  if (s_(r.instrucciones)) cambiosApto.instrucciones = s_(r.instrucciones);
  if (ap) actualizar_('Apartamentos', ap._fila, cambiosApto);
  else agregar_('Apartamentos', Object.assign({ torre: s_(r.torre), apartamento: s_(r.apartamento), activo: 'SI' }, cambiosApto));
  let resultado = { estatus: 'aprobada', renta_corta: false };
  // 2) Renta corta (Check-in): unidad, host y permiso exclusivo
  if (esSi_(r.es_renta_corta)) {
    if (!libroCheckin_()) throw error_('Configure CHECKIN_SHEET_ID para cargar la renta corta.');
    const codigo = prefijoTorre_(r.torre) + '-' + s_(r.apartamento).toUpperCase();
    const conflicto = conflictoAsignacion_(codigo, r.host_telefono);
    if (conflicto && conflicto.tipo === 'TEL_DUPLICADO') {
      actualizar_('Encuestas', r._fila, { estatus: 'conflicto', conflicto: conflicto.texto });
      throw error_(conflicto.texto + ' Corrija la pestaña Hosts antes de aprobar.', 'CONFLICTO');
    }
    if (conflicto && !req.transferir) {
      actualizar_('Encuestas', r._fila, { estatus: 'conflicto', conflicto: conflicto.texto });
      throw error_(conflicto.texto + ' Si corresponde, apruebe con «Transferir».', 'CONFLICTO');
    }
    const unidades = leerCI_('Unidades');
    if (!unidades.filas.some(u => s_(u.codigo_unidad).toUpperCase() === codigo)) {
      agregarCI_('Unidades', { codigo_unidad: codigo, torre: s_(r.torre), apartamento: s_(r.apartamento), estatus_unidad: 'activa', notas: 'Encuesta ' + s_(r.id) });
    }
    const hosts = leerCI_('Hosts').filas;
    let h = hosts.find(x => normTel_(x.telefono) === normTel_(r.host_telefono) && s_(x.estatus).toLowerCase() !== 'suspendido');
    if (!h) {
      const id = nuevoIdHost_(hosts);
      agregarCI_('Hosts', { id_host: id, nombre: s_(r.host_nombre), telefono: s_(r.host_telefono), correo: s_(r.host_correo), estatus: 'aprobado', notas: 'Encuesta ' + s_(r.id) });
      h = { id_host: id };
    }
    const asig = asignarUnidad_(codigo, s_(h.id_host), admin.nombre, !!req.transferir);
    resultado = { estatus: 'aprobada', renta_corta: true, codigo_unidad: codigo, id_host: s_(h.id_host), desactivados: asig.desactivados };
  }
  actualizar_('Encuestas', r._fila, { estatus: 'aprobada', revisado_por: admin.nombre, fecha_revision: new Date(), conflicto: '' });
  return resultado;
}

function accionTablero_() {
  const ahora = new Date(), hoy = dia_(ahora);
  const cats = {}; leer_('Categorias').filas.forEach(c => { cats[s_(c.codigo)] = { nombre: s_(c.nombre), letra: s_(c.letra), color: s_(c.color), trabajo: esSi_(c.es_personal_trabajo) }; });
  const torres = {};
  const T = n => (torres[n] = torres[n] || { torre: n, dentro: {}, visitas_adentro: 0, trabajadores_adentro: 0, trabajadores_tarde: 0,
    huespedes_adentro: 0, unidades_ocupadas: 0, unidades_renta: 0, llegadas_hoy: 0, llegadas_pendientes: 0, salidas_hoy: 0,
    salidas_vencidas: 0, entradas_hoy: 0, faltas_7d: 0, apartamentos: 0, encuestas_pendientes: 0, por_hora: new Array(24).fill(0), aptos_capacidad: [] });
  leer_('Torres').filas.forEach(t => { if (!esNo_(t.activa)) T(s_(t.torre)); });
  leer_('Lobbies').filas.forEach(l => { if (esSi_(l.activo)) T(s_(l.torre)); });
  leer_('Apartamentos').filas.forEach(a => { if (!esNo_(a.activo) && s_(a.torre)) T(s_(a.torre)).apartamentos++; });
  // Visitantes
  leer_('Visitas', FILAS_RECIENTES).filas.forEach(v => {
    const t = T(s_(v.torre)), c = s_(v.categoria);
    if (dia_(v.entrada) === hoy) { t.entradas_hoy++; t.por_hora[Number(Utilities.formatDate(fecha_(v.entrada), TZ, 'H'))]++; }
    if (s_(v.estatus) !== 'dentro') return;
    const n = 1 + (Number(v.acompanantes) || 0);
    t.dentro[c] = (t.dentro[c] || 0) + n;
    if (c === 'HUE') t.huespedes_lobby = (t.huespedes_lobby || 0) + n;
    if (cats[c] && cats[c].trabajo) { t.trabajadores_adentro += n; if (v.vence && fecha_(v.vence) < ahora) t.trabajadores_tarde += n; }
    else if (c !== 'HUE') t.visitas_adentro += n;
  });
  const hace7 = new Date(ahora.getTime() - 7 * 86400000);
  leer_('Incidencias').filas.forEach(i => { if (fecha_(i.fecha) >= hace7 && s_(i.estatus) !== 'anulada') T(s_(i.torre)).faltas_7d++; });
  leer_('Encuestas').filas.forEach(e => { if (['pendiente', 'conflicto'].indexOf(s_(e.estatus)) >= 0) T(s_(e.torre)).encuestas_pendientes++; });
  // Renta corta
  const hayCI = !!libroCheckin_();
  if (hayCI) {
    const unidades = leerCI_('Unidades').filas, hs = horaSalidaCI_();
    unidades.forEach(u => { if (s_(u.estatus_unidad || 'activa').toLowerCase() !== 'inactiva' && s_(u.torre)) T(s_(u.torre)).unidades_renta++; });
    const ocupadas = {};
    leerCI_('Reservas').filas.forEach(r => {
      const u = torreDeUnidad_(unidades, r.codigo_unidad); if (!u) return;
      const t = T(u.torre), est = s_(r.estatus), fin = finEstadia_(r, hs);
      if (est === 'ingresado') {
        if (fin && fin < ahora) t.salidas_vencidas++;
        else { t.huespedes_adentro += Number(r.num_personas) || 1; ocupadas[s_(r.codigo_unidad)] = u.torre; }
      }
      if (est === 'pendiente' && dia_(r.llegada) === hoy) { t.llegadas_hoy++; t.llegadas_pendientes++; }
      if (est === 'ingresado' && dia_(r.llegada) === hoy) t.llegadas_hoy++;
      if (['ingresado', 'finalizado'].indexOf(est) >= 0 && dia_(r.salida) === hoy) t.salidas_hoy++;
    });
    Object.keys(ocupadas).forEach(k => { T(ocupadas[k]).unidades_ocupadas++; });
    // El mismo huésped puede estar en el Check-in y en Visitantes: se toma el mayor, nunca la suma
    Object.values(torres).forEach(t => { t.dentro.HUE = Math.max(t.huespedes_lobby || 0, t.huespedes_adentro); t.huespedes_adentro = t.dentro.HUE; });
  }
  // Actividad reciente: bitácoras de los dos sistemas
  const feed = [];
  const bit = leer_('Bitacora', 60).filas;
  bit.forEach(b => feed.push({ fecha: b.fecha_hora, origen: 'Lobby', quien: s_(b.portero) || s_(b.lobby), accion: s_(b.accion), detalle: s_(b.detalle) }));
  if (hayCI) {
    const bci = leerCI_('Bitacora').filas.slice(-60);
    bci.forEach(b => feed.push({ fecha: b.fecha_hora, origen: 'Check-in', quien: s_(b.actor), accion: s_(b.accion), detalle: s_(b.detalle) }));
  }
  feed.sort((a, b) => fecha_(b.fecha) - fecha_(a.fecha));
  const lista = Object.values(torres).filter(t => t.torre).sort((a, b) => a.torre.localeCompare(b.torre));
  lista.forEach(t => { try { t.aptos_capacidad = alertasCapacidad_(t.torre); } catch (err) { t.aptos_capacidad = []; } });
  return { generado: ahora, torres: lista, categorias: cats, checkin_conectado: hayCI,
    conflictos: hayCI ? auditarConflictos_() : [], actividad: feed.slice(0, 40), version: VERSION_SERVICIO };
}

// Panel personal del host (enlace privado con token)
// ---------------------------------------------------------------------------
function accionPanelHost_(req) {
  if (!libroCheckin_()) throw error_('El panel no está disponible todavía.');
  const k = s_(req.k);
  const h = hostPorToken_(k);
  if (!h) throw error_('Este enlace no es válido o venció. Pida uno nuevo con /panel en el bot.', 'SIN_HOST');
  const idHost = s_(h.id_host), ahora = new Date(), hs = horaSalidaCI_();
  const unidades = leerCI_('Unidades').filas;
  const permisos = leerCI_('Permisos').filas.filter(p => esSi_(p.activo));
  const mias = permisos.filter(p => s_(p.id_host) === idHost).map(p => s_(p.codigo_unidad).toUpperCase());
  const compartidas = {}; permisos.forEach(p => { const c = s_(p.codigo_unidad).toUpperCase(); (compartidas[c] = compartidas[c] || new Set()).add(s_(p.id_host)); });
  const deuda = {}; leerCI_('Carga_CxC').filas.forEach(c => { if (s_(c.estatus_cuenta).toLowerCase().indexOf('venc') === 0) deuda[s_(c.codigo_unidad).toUpperCase()] = true; });
  const reservas = leerCI_('Reservas').filas.filter(r => s_(r.id_host) === idHost);
  const ini = new Date(ahora.getFullYear(), ahora.getMonth(), 1), fin = new Date(ahora.getFullYear(), ahora.getMonth() + 1, 1);
  const diasMes = Math.round((fin - ini) / 86400000);
  const noches = (r, a, b) => { const l = fecha_(r.llegada), s = fecha_(r.salida); if (!l || !s) return 0;
    const x = Math.max(l.getTime(), a.getTime()), y = Math.min(s.getTime(), b.getTime()); return Math.max(0, Math.round((y - x) / 86400000)); };
  const unidadesOut = Array.from(new Set(mias)).map(c => {
    const u = torreDeUnidad_(unidades, c) || { torre: '', apto: c, activa: true };
    const rs = reservas.filter(r => s_(r.codigo_unidad).toUpperCase() === c && s_(r.estatus) !== 'cancelado');
    const adentro = rs.find(r => s_(r.estatus) === 'ingresado' && (!finEstadia_(r, hs) || finEstadia_(r, hs) >= ahora));
    const ocup = rs.reduce((acc, r) => acc + noches(r, ini, fin), 0);
    let estado = 'libre';
    if (compartidas[c] && compartidas[c].size > 1) estado = 'en_revision';
    else if (deuda[c]) estado = 'bloqueada_deuda';
    else if (adentro) estado = 'ocupada';
    return { codigo: c, torre: u.torre, apto: u.apto, estado: estado, ficha: fichaPublica_(infoUnidadCI_(c), true), huesped: adentro ? s_(adentro.huesped_principal) : '',
      personas: adentro ? Number(adentro.num_personas) || 1 : 0, sale: adentro ? adentro.salida : '', noches_mes: ocup, ocupacion_maxima: capacidadUnidad_(c),
      ocupacion_mes: Math.min(100, Math.round(ocup / diasMes * 100)) };
  });
  const vista = r => ({ codigo: s_(r.codigo_reserva), unidad: s_(r.codigo_unidad), huesped: s_(r.huesped_principal), personas: Number(r.num_personas) || 1,
    llegada: r.llegada, salida: r.salida, estatus: s_(r.estatus), ingreso: r.ingreso_en || '', valorada: esSi_(r.valorada) });
  const hoyD = new Date(dia_(ahora) + 'T00:00:00');
  const proximas = reservas.filter(r => s_(r.estatus) === 'pendiente' && fecha_(r.llegada) >= hoyD).sort((a, b) => fecha_(a.llegada) - fecha_(b.llegada)).slice(0, 20).map(vista);
  const hace60 = new Date(ahora.getTime() - 60 * 86400000);
  const historial = reservas.filter(r => s_(r.estatus) !== 'pendiente' && fecha_(r.llegada) >= hace60).sort((a, b) => fecha_(b.llegada) - fecha_(a.llegada)).slice(0, 30).map(vista);
  const vals = leerCI_('Valoraciones').filas.filter(v => s_(v.id_host) === idHost);
  const prom = vals.length ? vals.reduce((a, v) => a + (Number(v.estrellas) || 0), 0) / vals.length : 0;
  const pendValorar = reservas.filter(r => ['ingresado', 'finalizado'].indexOf(s_(r.estatus)) >= 0 && !esSi_(r.valorada) && fecha_(r.salida) <= ahora).length;
  const reglamentos = {};
  unidadesOut.forEach(u => { if (u.torre && !reglamentos[u.torre]) reglamentos[u.torre] = normasPublicas_(u.torre, 'porteria').slice(0, 80); });
  return { host: { nombre: s_(h.nombre), id: idHost }, generado: ahora, unidades: unidadesOut, proximas: proximas, historial: historial, reglamentos: reglamentos, temas: TEMAS_NORMA,
    resumen: { reservas_mes: reservas.filter(r => s_(r.estatus) !== 'cancelado' && fecha_(r.llegada) >= ini && fecha_(r.llegada) < fin).length,
      noches_mes: unidadesOut.reduce((a, u) => a + u.noches_mes, 0), dias_mes: diasMes, valoraciones: vals.length, promedio_estrellas: Math.round(prom * 10) / 10,
      por_valorar: pendValorar } };
}

// ---------------------------------------------------------------------------
// Confirmar ingreso de un huésped de renta corta desde la app del lobby
// (reemplaza a AppSheet: el bot ve el estatus «ingresado» y avisa al host)
// ---------------------------------------------------------------------------
function accionCheckinConfirmar_(req, lobby) {
  if (!libroCheckin_()) throw error_('Configure CHECKIN_SHEET_ID para confirmar huéspedes.');
  const codigo = s_(req.codigo_reserva).toUpperCase();
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const r = leerCI_('Reservas').filas.find(x => s_(x.codigo_reserva).toUpperCase() === codigo);
    if (!r) throw error_('La reserva ' + codigo + ' no existe.');
    const u = torreDeUnidad_(leerCI_('Unidades').filas, r.codigo_unidad);
    if (u && u.torre && !mismaTorre_(u.torre, lobby.torre)) throw error_('La reserva es para ' + u.torre + '.');
    if (s_(r.estatus) === 'cancelado') throw error_('La reserva ' + codigo + ' está CANCELADA.');
    const personasReq = (req.personas || []).filter(p => s_(p.documento));
    if (s_(r.estatus) === 'ingresado' && !personasReq.length) return { ya_estaba: true };
    const inicio = inicioEstadia_(r);
    if (inicio && inicio > new Date()) {
      throw error_('QR todavía no válido: el apartamento recibe desde el ' + Utilities.formatDate(inicio, TZ, 'dd/MM/yyyy') +
        ' a las ' + Utilities.formatDate(inicio, TZ, 'h:mm a') + '. No dar acceso antes.', 'ANTES_DE_HORA');
    }
    const fin = finEstadia_(r, horaSalidaCI_());
    if (fin && fin < new Date()) throw error_('La reserva ' + codigo + ' ya terminó.');
    const personas = personasReq;
    if (!personas.length) throw error_('Cada persona que entra debe registrarse con su documento. Escanee al menos el del huésped principal.', 'REGISTRO_INDIVIDUAL');
    const vetada = personas.map(p => ({ p: p, v: vetado_(p.documento, lobby.torre) })).find(x => x.v);
    if (vetada) {
      registrar_(lobby, req.portero, 'intento_lista_negra', codigo + ' ' + s_(vetada.p.documento));
      throw error_('ACCESO DENEGADO: ' + s_(vetada.p.nombre || vetada.p.documento) + ' está en la lista negra. ' + vetada.v.motivo, 'VETADO');
    }
    const filasPer = leerCI_('Personas').filas.filter(p => s_(p.codigo_reserva).toUpperCase() === codigo);
    const enLobby = filasPer.filter(p => s_(p.origen) === 'lobby' || p.hora_ingreso).map(p => normDoc_(p.numero_documento));
    const nuevas = personas.filter((p, i) => enLobby.indexOf(normDoc_(p.documento)) < 0 && personas.findIndex(q => normDoc_(q.documento) === normDoc_(p.documento)) === i);
    if (!nuevas.length) return { ya_estaba: true };
    const declaradas = Math.max(Number(r.num_personas) || 1, 1);
    if (enLobby.length + nuevas.length > declaradas) {
      throw error_('La reserva es para ' + declaradas + ' persona(s) y ya ingresaron ' + enLobby.length + '. No dar acceso a más personas: el host debe registrar una reserva nueva o corregirla con la administración.', 'MAS_PERSONAS');
    }
    // Ocupación máxima del apartamento
    const apto = u ? u.apto : '';
    let capacidad = null;
    if (apto) {
      const oc = ocupacionApto_(lobby.torre, apto);
      const total = oc.residentes + oc.visitas + Math.max(oc.hue_visitas, oc.reserva_personas + nuevas.length);
      if (oc.max && total > oc.max && !s_(req.autoriza_capacidad)) {
        throw error_('El apartamento ' + apto + ' admite ' + oc.max + ' personas y quedaría con ' + total + '. No dar acceso sin autorización de la administración.', 'CAPACIDAD');
      }
      capacidad = { apartamento: apto, total: total, max: oc.max, llena: !!oc.max && total >= oc.max, excedida: !!oc.max && total > oc.max };
      if (capacidad.excedida) avisarCapacidadExcedida_(lobby, req.portero, apto, total, oc.max, s_(req.autoriza_capacidad));
      else if (capacidad.llena) registrar_(lobby, req.portero, 'alerta_capacidad', 'Apto ' + apto + ' en capacidad máxima (' + total + '/' + oc.max + ')');
    }
    const yaReg = filasPer.map(p => normDoc_(p.numero_documento));
    nuevas.forEach((p, i) => {
      const fila = filasPer.find(x => normDoc_(x.numero_documento) === normDoc_(p.documento));
      if (fila) { actualizarCI_('Personas', fila._fila, { hora_ingreso: new Date(), portero: s_(req.portero) }); return; }
      agregarCI_('Personas', { id_persona: codigo + '-L' + (i + 1) + aleatorio_(2), codigo_reserva: codigo, nombre: s_(p.nombre).toUpperCase(),
        tipo_documento: s_(p.tipo_documento), numero_documento: s_(p.documento).toUpperCase(), rol: yaReg.length || i ? 'acompanante' : 'principal',
        origen: 'lobby', hora_ingreso: new Date(), portero: s_(req.portero) });
    });
    if (s_(r.estatus) !== 'ingresado') actualizarCI_('Reservas', r._fila, { estatus: 'ingresado', ingreso_en: new Date(), portero: s_(req.portero) });
    registrar_(lobby, req.portero, 'checkin_confirmado', codigo + ' · ' + nuevas.length + ' persona(s)');
    return { confirmado: true, unidad: s_(r.codigo_unidad), personas_nuevas: nuevas.length, capacidad: capacidad };
  } finally { lock.releaseLock(); }
}

function horaTexto_(v) {
  if (v === '' || v === null || v === undefined) return '';
  if (v instanceof Date) return ('0' + v.getHours()).slice(-2) + ':' + ('0' + v.getMinutes()).slice(-2);
  if (typeof v === 'number') { const m = Math.round((v % 1) * 1440); return ('0' + Math.floor(m / 60)).slice(-2) + ':' + ('0' + m % 60).slice(-2); }
  const m = s_(v).match(/^(\d{1,2}):(\d{2})/);
  return m ? ('0' + m[1]).slice(-2) + ':' + m[2] : '';
}

const CAMPOS_FICHA = ['hora_entrada', 'hora_salida', 'codigo_puerta', 'wifi_red', 'wifi_clave', 'parqueo', 'ubicacion', 'mapa_url', 'normas']
  .concat(PARAMS_UNIDAD);
/** Valida y normaliza los parámetros que fija el host (noches, huéspedes, mascotas, fumar, fiestas). */
function validarParametros_(d, codigo) {
  const ent = (k, max) => {
    if (!(k in d) || s_(d[k]) === '') return '';
    const n = Number(s_(d[k]));
    if (!Number.isInteger(n) || n < 0 || n > max) throw error_('«' + k.replace('_', ' ') + '» debe ser un número entero entre 0 y ' + max + '.', 'PARAMETRO');
    return n || '';
  };
  const out = {};
  ['minimo_noches', 'maximo_noches'].forEach(k => { if (k in d) out[k] = ent(k, 365); });
  if ('max_huespedes' in d) out.max_huespedes = ent('max_huespedes', 50);
  if (out.minimo_noches && out.maximo_noches && out.minimo_noches > out.maximo_noches)
    throw error_('Las noches mínimas no pueden ser más que las máximas.', 'PARAMETRO');
  const cap = capacidadUnidad_(codigo);
  if (out.max_huespedes && cap && out.max_huespedes > cap)
    throw error_('La administración fijó para esta unidad un máximo de ' + cap + ' personas. Su límite debe ser ' + cap + ' o menos.', 'CAPACIDAD');
  ['mascotas', 'fumar', 'fiestas'].forEach(k => {
    if (!(k in d)) return;
    const v = s_(d[k]).toUpperCase();
    out[k] = v === 'SI' || v === 'SÍ' ? 'SI' : v === 'NO' ? 'NO' : '';
  });
  return out;
}
/** Límite efectivo de personas: el menor entre el de la administración y el del host (0 = sin límite). */
function limitePersonasUnidad_(codigo) {
  const cap = capacidadUnidad_(codigo), host = Number(infoUnidadCI_(codigo).max_huespedes) || 0;
  return cap && host ? Math.min(cap, host) : (cap || host);
}
function fichaPublica_(info, incluirSensibles) {
  const o = {};

    let v = k.indexOf('hora_') === 0 ? horaTexto_(info[k]) : s_(info[k]);
    if (!incluirSensibles && CAMPOS_SENSIBLES.indexOf(k) >= 0) v = v ? '••••' : '';
    o[k] = v;
  });
  o.actualizado_en = info.actualizado_en || '';
  return o;
}
function hostPorToken_(k) {

/** El host guarda la ficha de SU unidad. Solo si es el único host activo de esa unidad. */
function accionUnidadGuardar_(req) {
  if (!libroCheckin_()) throw error_('El panel no está disponible todavía.');
  const h = hostPorToken_(req.k);
  if (!h) throw error_('Este enlace no es válido o venció. Pida uno nuevo con /panel en el bot.', 'SIN_HOST');
  const codigo = s_(req.codigo_unidad).toUpperCase();
  const activos = leerCI_('Permisos').filas.filter(p => esSi_(p.activo) && s_(p.codigo_unidad).toUpperCase() === codigo);
  const ids = Array.from(new Set(activos.map(p => s_(p.id_host))));
  if (ids.indexOf(s_(h.id_host)) < 0) throw error_('Esa unidad no está asignada a usted.');
  if (ids.length > 1) throw error_('La unidad está en revisión (dos hosts). Stili debe resolverlo antes de editar (C-25).', 'CONFLICTO');
  const d = req.ficha || {};
  ['hora_entrada', 'hora_salida'].forEach(k => { if (s_(d[k]) && !horaTexto_(d[k])) throw error_('La hora debe tener el formato 15:00.'); });
  if (s_(d.mapa_url) && !/^https:\/\//i.test(s_(d.mapa_url))) throw error_('El enlace del mapa debe empezar con https://');
  const cambios = validarParametros_(d, codigo);
  CAMPOS_FICHA.forEach(k => {
    if (!(k in d) || PARAMS_UNIDAD.indexOf(k) >= 0) return;
    let v = s_(d[k]).slice(0, k === 'normas' ? 2000 : 200);
    if (k.indexOf('hora_') === 0) v = horaTexto_(v);
    cambios[k] = v;
  });
  cambios.actualizado_por = s_(h.id_host); cambios.actualizado_en = new Date();
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const fila = leerCI_('Unidades_Info').filas.find(r => s_(r.codigo_unidad).toUpperCase() === codigo);
    if (fila) actualizarCI_('Unidades_Info', fila._fila, cambios);
    else agregarCI_('Unidades_Info', Object.assign({ codigo_unidad: codigo }, cambios));
  } finally { lock.releaseLock(); }
  registrar_({ nombre: 'panel' }, s_(h.id_host), 'ficha_unidad', codigo + ' (' + Object.keys(cambios).filter(k => CAMPOS_FICHA.indexOf(k) >= 0).join(', ') + ')');
  return { guardado: true, ficha: fichaPublica_(infoUnidadCI_(codigo), true), ocupacion_maxima: capacidadUnidad_(codigo) };
}

function hashPin_(pin, id) {
  const b = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, s_(pin) + '|' + s_(id) + '|stili');
  return b.map(x => ('0' + (x & 0xff).toString(16)).slice(-2)).join('');
}

  const vetados = bit.filter(b => s_(b.accion) === 'intento_lista_negra' && deEsteLobby(b)).map(b => ({ hora: b.fecha_hora, detalle: s_(b.detalle) }));
  const alarmas = bit.filter(b => s_(b.accion) === 'alerta_cierre' && deEsteLobby(b)).length;
  const largos = dentro_(torre).filter(v => fecha_(v.entrada) < new Date(ahora.getTime() - 8 * 3600000)).map(v => ({ nombre: v.nombre, apto: v.destino_apto, entrada: v.entrada, categoria: v.categoria }));
  const entradas = leer_('Visitas', FILAS_RECIENTES).filas.filter(v => mismaTorre_(v.torre, torre) && fecha_(v.entrada) >= desde).length;
  const atender = [];
  const cita = k => hoy.respaldo && hoy.respaldo[k] ? ' (' + hoy.respaldo[k].cita + ')' : '';
  if (hoy.trabajadores_tarde.length) atender.push(hoy.trabajadores_tarde.length + ' trabajador(es) adentro fuera de hora' + cita('horario_trabajo'));
  if (hoy.salidas_vencidas.length) atender.push(hoy.salidas_vencidas.length + ' salida(s) de huéspedes vencida(s)' + cita('salida_huesped'));
  if (vetados.length) atender.push(vetados.length + ' intento(s) de entrada de personas en lista negra');
  if (faltas.some(f => f.gravedad === 'grave')) atender.push('Faltas graves registradas en el turno');
  if (largos.length) atender.push(largos.length + ' persona(s) adentro desde hace más de 8 horas');
  if (hoy.proximas.length) atender.push(hoy.proximas.length + ' llegada(s) de huéspedes en las próximas 2 horas');
  return { generado: ahora, desde: desde, atender: atender, adentro: hoy.adentro, total_adentro: hoy.total_adentro,
    trabajadores_tarde: hoy.trabajadores_tarde, salidas_vencidas: hoy.salidas_vencidas, proximas: hoy.proximas, llegadas: hoy.llegadas,
    invitaciones: hoy.invitaciones, faltas: faltas, lista_negra: vetados, alarmas_cierre: alarmas, mas_de_8h: largos, entradas_turno: entradas, respaldo: hoy.respaldo };
}
function empleadoPorPin_(req, torre) {
  const doc = normDoc_(req.documento), pin = s_(req.pin);
  if (!doc || !/^\d{4,6}$/.test(pin)) throw error_('Pase su cédula por el lector y escriba su PIN de 4 dígitos.');
  const e = leer_('Empleados').filas.find(x => normDoc_(x.documento) === doc && esSi_(x.activo));
  if (!e || hashPin_(pin, e.id) !== s_(e.pin_hash)) throw error_('Cédula o PIN incorrectos.', 'PIN');
  const vetoE = vetado_(doc, torre);
  if (vetoE) { registrar_({ nombre: torre }, s_(e.nombre), 'intento_lista_negra', 'Ponche de empleado vetado ' + doc); throw error_('ACCESO DENEGADO: este empleado está en la lista negra. Avise a la administración.', 'VETADO'); }
  if (s_(e.torre) && s_(e.torre).toUpperCase() !== 'TODAS' && !mismaTorre_(e.torre, torre)) throw error_('Usted no está asignado(a) a ' + torre + '.');
  return e;
}
function accionPonchar_(req, lobby) {
  const tipo = s_(req.tipo) === 'salida' ? 'salida' : 'entrada';
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const e = empleadoPorPin_(req, lobby.torre);
    const abiertos = turnoActual_(lobby.torre);
    const yo = abiertos.find(a => a.id === s_(e.id));
    if (tipo === 'entrada' && yo) throw error_('Usted ya tiene la entrada marcada desde ' + hora_(yo.desde) + '.');
    if (tipo === 'salida' && !yo) throw error_('No tiene una entrada marcada en esta torre.');
    const ahora = new Date();
    agregar_('Ponches', { id: 'PON-' + Utilities.formatDate(ahora, TZ, 'yyMMddHHmm') + '-' + aleatorio_(3), fecha_hora: ahora, torre: lobby.torre,
      id_empleado: s_(e.id), nombre: s_(e.nombre), tipo: tipo, lobby: lobby.nombre, notas: s_(req.notas).slice(0, 1000) });
    registrar_(lobby, s_(e.nombre), 'ponche_' + tipo, s_(e.id));
    if (tipo === 'salida') {
      const horas = (ahora - fecha_(yo.desde)) / 3600000;
      // Las notas del que sale quedan guardadas para el próximo relevo
      return { tipo: tipo, nombre: s_(e.nombre), horas: Math.round(horas * 100) / 100 };
    }
    // ENTRADA: resumen de relevo para quien llega
    const ultimaSalida = leer_('Ponches', 3000).filas.filter(p => mismaTorre_(p.torre, lobby.torre) && s_(p.tipo) === 'salida' && s_(p.id_empleado) !== s_(e.id))
      .sort((a, b) => fecha_(b.fecha_hora) - fecha_(a.fecha_hora))[0];
    const relevoPrevio = leer_('Relevos').filas.filter(r => mismaTorre_(r.torre, lobby.torre)).sort((a, b) => fecha_(b.fecha) - fecha_(a.fecha))[0];
    const desde = relevoPrevio ? fecha_(relevoPrevio.fecha) : new Date(ahora.getTime() - 12 * 3600000);
    const sale = abiertos.length ? abiertos.map(a => a.nombre).join(', ') : (ultimaSalida ? s_(ultimaSalida.nombre) : '');
    const notas = leer_('Ponches', 3000).filas.filter(p => mismaTorre_(p.torre, lobby.torre) && s_(p.tipo) === 'salida' && fecha_(p.fecha_hora) >= desde && s_(p.notas))
      .map(p => s_(p.nombre) + ': ' + s_(p.notas)).join('\n');
    const resumen = resumenTurno_(lobby.torre, desde);
    const id = 'REL-' + Utilities.formatDate(ahora, TZ, 'yyMMddHHmm') + '-' + aleatorio_(3);
    agregar_('Relevos', { id: id, torre: lobby.torre, fecha: ahora, sale: sale, entra: s_(e.nombre), resumen: JSON.stringify(resumen).slice(0, 45000),
      notas_saliente: notas, confirmado: 'NO' });
    return { tipo: tipo, nombre: s_(e.nombre), relevo: relevoPublico_(leer_('Relevos').filas.find(r => s_(r.id) === id)) };
  } finally { lock.releaseLock(); }
}
function accionRelevoConfirmar_(req, lobby) {
  const r = leer_('Relevos').filas.find(x => s_(x.id) === s_(req.id) && mismaTorre_(x.torre, lobby.torre));
  if (!r) throw error_('No encontré ese relevo.');
  actualizar_('Relevos', r._fila, { confirmado: 'SI', confirmado_en: new Date() });
  registrar_(lobby, s_(r.entra), 'relevo_confirmado', s_(r.id));
  return { confirmado: true };
}

function accionEmpleadoGuardar_(req, admin) {
  const d = req.empleado || {};
  if (!s_(d.nombre) || !normDoc_(d.documento)) throw error_('Escriba nombre y cédula del empleado.');
  if (s_(d.pin) && !/^\d{4,6}$/.test(s_(d.pin))) throw error_('El PIN debe tener de 4 a 6 números.');
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const todos = leer_('Empleados').filas;
    const repetido = todos.find(e => normDoc_(e.documento) === normDoc_(d.documento) && s_(e.id) !== s_(d.id));
    if (repetido) throw error_('Ya existe un empleado con esa cédula (' + s_(repetido.nombre) + ').');
    const vetoE = vetado_(d.documento, '');
    if (vetoE) throw error_('No se puede registrar: la cédula está en la lista negra (' + vetoE.motivo + ').', 'VETADO');
    let e = s_(d.id) && todos.find(x => s_(x.id) === s_(d.id));
    if (!e) {
      const id = 'EMP-' + ('00' + (todos.length + 1)).slice(-3) + aleatorio_(2);
      if (!s_(d.pin)) throw error_('Asigne un PIN de 4 dígitos al empleado nuevo.');
      agregar_('Empleados', { id: id, nombre: s_(d.nombre).toUpperCase(), documento: s_(d.documento), torre: s_(d.torre) || 'TODAS', rol: s_(d.rol) || 'portero',
        pin_hash: hashPin_(d.pin, id), activo: 'SI', creado_en: new Date() });
      registrar_({ nombre: 'admin' }, admin.nombre, 'empleado_nuevo', id);
      return { id: id };
    }
    const cambios = { nombre: s_(d.nombre).toUpperCase(), torre: s_(d.torre) || 'TODAS', rol: s_(d.rol) || 'portero', activo: d.activo === false ? 'NO' : 'SI' };
    if (s_(d.pin)) cambios.pin_hash = hashPin_(d.pin, e.id);
    actualizar_('Empleados', e._fila, cambios);
    registrar_({ nombre: 'admin' }, admin.nombre, 'empleado_editado', s_(e.id));
    return { id: s_(e.id) };
  } finally { lock.releaseLock(); }
}
function accionAsistencia_(req) {
  const desde = s_(req.desde) ? new Date(s_(req.desde) + 'T00:00:00') : new Date(Date.now() - 7 * 86400000);
  const hasta = s_(req.hasta) ? new Date(s_(req.hasta) + 'T23:59:59') : new Date();
  const ps = leer_('Ponches', 20000).filas.filter(p => fecha_(p.fecha_hora) >= new Date(desde.getTime() - 86400000) && fecha_(p.fecha_hora) <= hasta)
    .sort((a, b) => fecha_(a.fecha_hora) - fecha_(b.fecha_hora));
  const abiertos = {}, turnos = [];
  ps.forEach(p => {
    const k = s_(p.id_empleado) + '|' + s_(p.torre);
    if (s_(p.tipo) === 'entrada') abiertos[k] = p;
    else if (abiertos[k]) { const a = abiertos[k]; delete abiertos[k];
      if (fecha_(a.fecha_hora) >= desde) turnos.push({ nombre: s_(a.nombre), torre: s_(a.torre), entrada: a.fecha_hora, salida: p.fecha_hora,
        horas: Math.round((fecha_(p.fecha_hora) - fecha_(a.fecha_hora)) / 36000) / 100 }); }
  });
  Object.values(abiertos).forEach(a => { if (fecha_(a.fecha_hora) >= desde) turnos.push({ nombre: s_(a.nombre), torre: s_(a.torre), entrada: a.fecha_hora, salida: '', horas: null }); });
  const totales = {};
  turnos.forEach(t => { totales[t.nombre] = (totales[t.nombre] || 0) + (t.horas || 0); });
  return { turnos: turnos.sort((a, b) => fecha_(b.entrada) - fecha_(a.entrada)),
    totales: Object.keys(totales).map(n => ({ nombre: n, horas: Math.round(totales[n] * 100) / 100 })).sort((a, b) => b.horas - a.horas) };
}

function soloGerente_(fn) {
  return (req, admin) => {

    return fn(req, admin);

function temaSugerido_(texto) {
  let mejor = 'otro', puntos = 0;
  Object.keys(PALABRAS_TEMA).forEach(k => {
    const n = (s_(texto).match(PALABRAS_TEMA[k]) || []).length;
    if (n > puntos) { puntos = n; mejor = k; }
  });
  return mejor;
}
/** Divide el texto en artículos. Reconoce «Artículo 12», «ARTÍCULO 12.-», «Art. 12:», «Artículo Único». */
function dividirArticulos_(texto) {
  const limpio = s_(texto).replace(/\r/g, '').replace(/[ \t]+/g, ' ').replace(/\n{3,}/g, '\n\n');
  const re = /(^|\n)\s*((?:art[íi]culo|art\.)\s*(\d+[a-z]?|[úu]nico|primero|segundo|tercero|cuarto|quinto|sexto|s[ée]ptimo|octavo|noveno|d[ée]cimo)\b\s*[º°o]?\s*[\.\-–:)]*)/gi;
  const marcas = [];
  let m;
  while ((m = re.exec(limpio))) marcas.push({ i: m.index + m[1].length, fin: re.lastIndex, etiqueta: 'Artículo ' + m[3].replace(/^./, c => c.toUpperCase()) });
  const out = [];
  if (marcas.length) {
    marcas.forEach((mk, j) => {
      const cuerpo = limpio.slice(mk.fin, j + 1 < marcas.length ? marcas[j + 1].i : limpio.length).trim();
      if (cuerpo) out.push({ articulo: mk.etiqueta, texto: cuerpo.slice(0, 4000) });
    });
  } else {
    // Sin artículos: se agrupan los párrafos en secciones
    limpio.split(/\n\s*\n/).map(x => x.trim()).filter(x => x.length > 20).forEach((p, j) => out.push({ articulo: 'Sección ' + (j + 1), texto: p.slice(0, 4000) }));
  }
  return out.slice(0, 400);
}
function carpetaDocumentos_() {
  const cfg = config_();
  if (cfg.CARPETA_DOCUMENTOS_ID) {
    try { return DriveApp.getFolderById(cfg.CARPETA_DOCUMENTOS_ID); } catch (err) { /* se crea abajo */ }
  }
  const padre = DriveApp.getFileById(libro_().getId()).getParents();
  const carpeta = (padre.hasNext() ? padre.next() : DriveApp.getRootFolder()).createFolder('Documentos y normas');
  fijarConfig_('CARPETA_DOCUMENTOS_ID', carpeta.getId());
  return carpeta;
}
/** PDF o Word → texto con el servicio avanzado de Drive (si está activado). Si no, se usa el texto que envía el panel. */
function textoDeArchivo_(blob) {
  if (typeof Drive === 'undefined' || !Drive.Files) return '';
  const tmp = Drive.Files.create({ name: 'tmp-ocr-' + Date.now(), mimeType: 'application/vnd.google-apps.document' }, blob, { ocrLanguage: 'es' });
  try { return DocumentApp.openById(tmp.id).getBody().getText(); }
  finally { try { DriveApp.getFileById(tmp.id).setTrashed(true); } catch (e) { /* nada */ } }
}
function docAplica_(d, torre) {
  const t = s_(d.torre).toUpperCase();
  return !t || t === 'TODAS' || mismaTorre_(d.torre, torre);
}
/** Artículos activos para una torre. destino: 'porteria' | 'huesped' | '' (todos, para el bot y los avisos). */
function normasPublicas_(torre, destino) {
  const docs = {};
  leer_('Documentos').filas.forEach(d => {
    if (!esSi_(d.activo) || !docAplica_(d, torre)) return;
    if (destino === 'porteria' && esNo_(d.visible_porteria)) return;
    if (destino === 'huesped' && !esSi_(d.visible_huesped)) return;
    docs[s_(d.id)] = d;
  });
  return leer_('Normas').filas.filter(n => docs[s_(n.id_documento)] && !esNo_(n.activo))
    .sort((a, b) => (Number(a.orden) || 0) - (Number(b.orden) || 0))
    .map(n => { const d = docs[s_(n.id_documento)];
      return { id: s_(n.id), documento: s_(d.titulo), tipo: s_(d.tipo), torre: s_(d.torre) || 'TODAS', articulo: s_(n.articulo), tema: s_(n.tema) || 'otro', texto: s_(n.texto) }; });
}
/** El artículo que respalda un aviso: primero el de la torre, luego el general (TODAS). */
function respaldoNorma_(torre, tema) {
  const lista = normasPublicas_(torre, '').filter(n => n.tema === tema);
  if (!lista.length) return null;
  const propia = lista.find(n => mismaTorre_(n.torre, torre)) || lista[0];
  return { documento: propia.documento, articulo: propia.articulo, texto: recortar_(propia.texto, 420), cita: propia.documento + ', ' + propia.articulo };
}
function respaldosClave_(torre) {
  return { salida_huesped: respaldoNorma_(torre, 'salida_huesped'), horario_trabajo: respaldoNorma_(torre, 'horario_trabajo') };
}
function recortar_(t, n) { t = s_(t).replace(/\s+/g, ' '); return t.length > n ? t.slice(0, n - 1).replace(/\s+\S*$/, '') + '…' : t; }

function accionDocumentos_(req, admin) {
  const normas = leer_('Normas').filas;
  const docs = leer_('Documentos').filas.map(d => ({ id: s_(d.id), torre: s_(d.torre) || 'TODAS', titulo: s_(d.titulo), tipo: s_(d.tipo),
    archivo: s_(d.archivo_id) ? 'https://drive.google.com/file/d/' + s_(d.archivo_id) + '/view' : '', archivo_nombre: s_(d.archivo_nombre),
    visible_porteria: !esNo_(d.visible_porteria), visible_huesped: esSi_(d.visible_huesped), activo: esSi_(d.activo),
    subido_por: s_(d.subido_por), subido_en: d.subido_en,
    normas: normas.filter(n => s_(n.id_documento) === s_(d.id)).sort((a, b) => (Number(a.orden) || 0) - (Number(b.orden) || 0))
      .map(n => ({ id: s_(n.id), articulo: s_(n.articulo), tema: s_(n.tema) || 'otro', texto: s_(n.texto), activo: !esNo_(n.activo) })) }))
    .sort((a, b) => fecha_(b.subido_en) - fecha_(a.subido_en));
  const torres = leer_('Torres').filas.map(x => s_(x.torre)).filter(Boolean);
  const temasCubiertos = {};
  docs.filter(d => d.activo).forEach(d => d.normas.filter(n => n.activo).forEach(n => { temasCubiertos[n.tema] = (temasCubiertos[n.tema] || 0) + 1; }));
  return { documentos: docs, temas: TEMAS_NORMA, tipos: TIPOS_DOCUMENTO, torres: torres, temas_cubiertos: temasCubiertos, puede_editar: admin.rol === 'gerente',
    faltan: ['salida_huesped', 'horario_trabajo'].filter(k => !temasCubiertos[k]) };
}

function accionDocumentoSubir_(req, admin) {
  const titulo = s_(req.titulo).slice(0, 150);
  if (!titulo) throw error_('Escriba el título del documento.');
  const tipo = TIPOS_DOCUMENTO[s_(req.tipo)] ? s_(req.tipo) : 'otro';
  const torre = s_(req.torre).toUpperCase() === 'TODAS' || !s_(req.torre) ? 'TODAS' : s_(req.torre);
  if (torre !== 'TODAS' && !leer_('Torres').filas.some(x => mismaTorre_(x.torre, torre)) && !leer_('Lobbies').filas.some(x => mismaTorre_(x.torre, torre)))
    throw error_('La torre «' + torre + '» no existe.');
  let texto = s_(req.texto), archivoId = '', nombre = s_(req.nombre_archivo).slice(0, 120);
  if (req.base64) {
    const bytes = Utilities.base64Decode(s_(req.base64).replace(/^data:[^,]*,/, ''));
    if (bytes.length > 15 * 1024 * 1024) throw error_('El archivo pasa de 15 MB.', 'DOC_GRANDE');
    const mime = s_(req.mime) || 'application/pdf';
    if (!/pdf|word|officedocument|text\/plain/.test(mime)) throw error_('Formato no admitido. Use PDF, Word (.docx) o texto.', 'DOC_FORMATO');
    const blob = Utilities.newBlob(bytes, mime, nombre || titulo);
    const f = carpetaDocumentos_().createFile(blob);
    f.setSharing(DriveApp.Access.PRIVATE, DriveApp.Permission.NONE);
    archivoId = f.getId();
    if (!texto) { try { texto = textoDeArchivo_(blob); } catch (err) { texto = ''; } }
  }
  const articulos = dividirArticulos_(texto);
  if (!articulos.length) throw error_('No se pudo leer texto del documento. Si es un escaneo, pegue el texto en la casilla «Texto del documento» o active el servicio Drive (ver manual, falla V-28).', 'DOC_SIN_TEXTO');
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const id = 'DOC-' + aleatorio_(6);
    agregar_('Documentos', { id: id, torre: torre, titulo: titulo, tipo: tipo, archivo_id: archivoId, archivo_nombre: nombre,
      visible_porteria: req.visible_porteria === false ? 'NO' : 'SI', visible_huesped: req.visible_huesped ? 'SI' : 'NO',
      articulos: articulos.length, subido_por: admin.nombre, subido_en: new Date(), activo: 'SI', notas: s_(req.notas).slice(0, 300) });
    const sh = hoja_('Normas');
    const enc = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String);
    const filas = articulos.map((a, j) => { const o = { id: id + '-' + ('00' + (j + 1)).slice(-3), id_documento: id, torre: torre, articulo: a.articulo,
      tema: temaSugerido_(a.texto), texto: a.texto, activo: 'SI', orden: j + 1 }; return enc.map(k => celda_(o[k], k)); });
    sh.getRange(sh.getLastRow() + 1, 1, filas.length, enc.length).setValues(filas);
    registrar_({ nombre: 'admin' }, admin.nombre, 'documento_subido', id + ' · ' + titulo + ' · ' + articulos.length + ' artículos');
    try { sincronizarNormasCI_(); } catch (err) { /* el bot usará la copia anterior */ }
    return accionDocumentos_(req, admin);
  } finally { lock.releaseLock(); }
}

function accionDocumentoEstado_(req, admin) {
  const d = leer_('Documentos').filas.find(x => s_(x.id) === s_(req.id));
  if (!d) throw error_('Documento no encontrado.');
  const cambios = {};
  ['activo', 'visible_porteria', 'visible_huesped'].forEach(k => { if (typeof req[k] === 'boolean') cambios[k] = req[k] ? 'SI' : 'NO'; });
  if (req.torre) cambios.torre = s_(req.torre);
  if (req.titulo) cambios.titulo = s_(req.titulo).slice(0, 150);
  actualizar_('Documentos', d._fila, cambios);
  registrar_({ nombre: 'admin' }, admin.nombre, 'documento_cambiado', s_(d.id) + ' ' + JSON.stringify(cambios));
  try { sincronizarNormasCI_(); } catch (err) { /* nada */ }
  return accionDocumentos_(req, admin);
}

function accionNormaGuardar_(req, admin) {
  const n = leer_('Normas').filas.find(x => s_(x.id) === s_(req.id));
  if (!n) throw error_('Artículo no encontrado.');
  const cambios = {};
  if (req.tema !== undefined) { if (!TEMAS_NORMA[s_(req.tema)]) throw error_('Tema no válido.'); cambios.tema = s_(req.tema); }
  if (req.articulo !== undefined) cambios.articulo = s_(req.articulo).slice(0, 80);
  if (req.texto !== undefined) { if (!s_(req.texto)) throw error_('El texto no puede quedar vacío.'); cambios.texto = s_(req.texto).slice(0, 4000); }
  if (typeof req.activo === 'boolean') cambios.activo = req.activo ? 'SI' : 'NO';
  actualizar_('Normas', n._fila, cambios);
  registrar_({ nombre: 'admin' }, admin.nombre, 'norma_cambiada', s_(n.id) + ' ' + Object.keys(cambios).join(','));
  try { sincronizarNormasCI_(); } catch (err) { /* nada */ }
  return accionDocumentos_(req, admin);
}

/** Copia a la hoja del Check-in los artículos con tema (solo texto) para que el bot los cite. */
function sincronizarNormasCI_() {
  const ci = libroCheckin_();
  if (!ci) return 0;
  let sh = ci.getSheetByName('Normas');
  const enc = TABLAS_CHECKIN_NUEVAS.Normas;
  if (!sh) { sh = ci.insertSheet('Normas'); sh.getRange(1, 1, 1, enc.length).setValues([enc]).setFontWeight('bold'); sh.setFrozenRows(1); }
  const docs = {};
  leer_('Documentos').filas.filter(d => esSi_(d.activo)).forEach(d => { docs[s_(d.id)] = d; });
  const ahora = new Date();
  const filas = leer_('Normas').filas.filter(n => docs[s_(n.id_documento)] && !esNo_(n.activo) && s_(n.tema) && s_(n.tema) !== 'otro')
    .map(n => [s_(docs[s_(n.id_documento)].torre) || 'TODAS', s_(n.tema), s_(n.articulo), s_(docs[s_(n.id_documento)].titulo), recortar_(n.texto, 900), ahora]);
  if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, enc.length).clearContent();
  if (filas.length) sh.getRange(2, 1, filas.length, enc.length).setValues(filas);
  delete _cacheCI.Normas;
  return filas.length;
}


// ===========================================================================
// Ocupación máxima por apartamento
// Ocupación = residentes habituales + personas registradas adentro con ese destino + huéspedes de renta corta.
// Todo el que entra se registra uno por uno (REGISTRO_INDIVIDUAL, por defecto SI).
// ===========================================================================
function cuentaOcupacion_(cat) {
  const c = typeof cat === 'string' ? (leer_('Categorias').filas.find(x => s_(x.codigo) === cat) || { codigo: cat }) : (cat || {});
  if (s_(c.cuenta_ocupacion)) return !esNo_(c.cuenta_ocupacion);
  return s_(c.codigo) !== 'DEL';   // el delivery se queda en el lobby
}
function filaApto_(torre, apto) {
  return leer_('Apartamentos').filas.find(a => mismaTorre_(a.torre, torre) && s_(a.apartamento).toUpperCase() === s_(apto).toUpperCase()) || null;
}
function capacidadDe_(fila) {
  const propia = fila ? Number(fila.ocupacion_maxima) || 0 : 0;
  return propia || Number(config_().OCUPACION_MAXIMA_DEFECTO) || 0;
}
/** Capacidad de una unidad de renta corta (por su código TORRE-APTO). 0 = sin límite. */
function capacidadUnidad_(codigo) {
  if (!libroCheckin_()) return 0;
  const u = torreDeUnidad_(leerCI_('Unidades').filas, codigo);
  return u ? capacidadDe_(filaApto_(u.torre, u.apto)) : 0;
}
/** Ocupación de todos los apartamentos con gente o con residentes de una torre. */
function ocupacionTorre_(torre) {
  const out = {};
  const base = a => (out[a] = out[a] || { apartamento: a, residentes: 0, visitas: 0, hue_visitas: 0, reserva_personas: 0, max: 0, personas: [] });
  leer_('Apartamentos').filas.filter(a => mismaTorre_(a.torre, torre) && !esNo_(a.activo)).forEach(a => {
    const o = base(s_(a.apartamento).toUpperCase()); o.residentes = Number(a.residentes_habituales) || 0; o.max = capacidadDe_(a);
  });
  const cats = {}; leer_('Categorias').filas.forEach(c => { cats[s_(c.codigo)] = c; });
  dentro_(torre).forEach(v => {
    const ap = s_(v.destino_apto).toUpperCase(); if (!ap) return;
    if (!cuentaOcupacion_(cats[v.categoria] || v.categoria)) return;
    const o = base(ap), n = 1 + (Number(v.acompanantes) || 0);
    if (v.categoria === 'HUE') o.hue_visitas += n; else o.visitas += n;
    o.personas.push({ nombre: v.nombre, categoria: v.categoria, gafete: v.gafete });
  });
  if (libroCheckin_()) {
    const ahora = new Date(), hs = horaSalidaCI_();
    const per = leerCI_('Personas').filas;
    reservasDeTorre_(torre).forEach(x => {
      if (s_(x.r.estatus) !== 'ingresado') return;
      const fin = finEstadia_(x.r, hs); if (fin && fin < ahora) return;
      const cod = s_(x.r.codigo_reserva).toUpperCase();
      const enLobby = per.filter(p => s_(p.codigo_reserva).toUpperCase() === cod && (s_(p.origen) === 'lobby' || p.hora_ingreso)).length;
      base(s_(x.u.apto).toUpperCase()).reserva_personas += enLobby || Number(x.r.num_personas) || 1;
    });
  }
  Object.values(out).forEach(o => {
    o.huespedes = Math.max(o.hue_visitas, o.reserva_personas);
    o.total = o.residentes + o.visitas + o.huespedes;
    o.estado = !o.max ? 'sin_limite' : o.total > o.max ? 'excedida' : o.total === o.max ? 'llena' : o.total === o.max - 1 ? 'casi' : 'ok';
  });
  return out;
}
function ocupacionApto_(torre, apto) {
  const ap = s_(apto).toUpperCase();
  const o = ocupacionTorre_(torre)[ap];
  if (o) return o;
  const fila = filaApto_(torre, ap);
  return { apartamento: ap, residentes: fila ? Number(fila.residentes_habituales) || 0 : 0, visitas: 0, hue_visitas: 0, reserva_personas: 0, huespedes: 0,
    max: capacidadDe_(fila), personas: [], total: fila ? Number(fila.residentes_habituales) || 0 : 0, estado: capacidadDe_(fila) ? 'ok' : 'sin_limite' };
}
/** Apartamentos en capacidad máxima o por encima: alimentan la alarma del lobby. */
function alertasCapacidad_(torre) {
  return Object.values(ocupacionTorre_(torre)).filter(o => o.estado === 'llena' || o.estado === 'excedida')
    .map(o => ({ apartamento: o.apartamento, total: o.total, max: o.max, estado: o.estado, residentes: o.residentes, visitas: o.visitas, huespedes: o.huespedes }))
    .sort((a, b) => a.apartamento.localeCompare(b.apartamento, 'es', { numeric: true }));
}
function avisarCapacidadExcedida_(lobby, portero, apto, total, max, autoriza) {
  registrar_(lobby, portero, 'capacidad_excedida', 'Apto ' + apto + ': ' + total + '/' + max + ' · autorizó ' + (autoriza || '—'));
  const correo = config_().CORREO_ADMINISTRACION;
  if (correo) {
    try { MailApp.sendEmail(correo, 'Capacidad excedida · ' + lobby.torre + ' Apto ' + apto,
      'El apartamento ' + apto + ' de ' + lobby.torre + ' quedó con ' + total + ' personas (máximo ' + max + ').\nAutorizó: ' + (autoriza || '—') + '\nPortero: ' + s_(portero)); } catch (err) { /* nada */ }
  }
}
/** Panel: capacidad y residentes de un apartamento. Se copia a la hoja del Check-in para que el bot la respete. */
function accionApartamentoGuardar_(req, admin) {
  const torre = s_(req.torre), apto = s_(req.apartamento).toUpperCase();
  const fila = filaApto_(torre, apto);
  if (!fila) throw error_('El apartamento no existe en ' + torre + '.');
  const max = Number(req.ocupacion_maxima), res = Number(req.residentes_habituales);
  if (!(max >= 0 && max <= 50) || !(res >= 0 && res <= 50) || (max && res > max)) throw error_('Revise los números: la capacidad va de 0 (sin límite) a 50 y no puede ser menor que los residentes.');
  actualizar_('Apartamentos', fila._fila, { ocupacion_maxima: max || '', residentes_habituales: res || '' });
  sincronizarCapacidadCI_(torre, apto, max);
  registrar_({ nombre: 'admin' }, admin.nombre, 'capacidad_cambiada', torre + ' ' + apto + ': máx ' + (max || 'sin límite') + ', residentes ' + res);
  return { ocupacion: ocupacionApto_(torre, apto) };
}
function sincronizarCapacidadCI_(torre, apto, max) {
  if (!libroCheckin_()) return;
  const u = leerCI_('Unidades').filas.find(x => mismaTorre_(x.torre, torre) && s_(x.apartamento).toUpperCase() === s_(apto).toUpperCase());
  if (u && leerCI_('Unidades').enc.indexOf('ocupacion_maxima') >= 0) actualizarCI_('Unidades', u._fila, { ocupacion_maxima: max || '' });
}

// Reporte de conducta del huésped → el host lo recibe por Telegram (el bot lee Reportes_Conducta)
// ===========================================================================
const TEMA_INCIDENCIA = { 'Ruido o escándalo': 'ruido', 'Estacionamiento indebido': 'parqueo', 'Basura o suciedad': 'basura',
  'Entró a área no autorizada': 'areas_comunes', 'Daños a la propiedad': 'areas_comunes', 'Conducta irrespetuosa': 'sanciones',
  'Incumplió horario': 'horario_trabajo', 'No registró salida': 'visitas' };
function reservaDeHuesped_(d, torre) {
  if (!libroCheckin_()) return null;
  const reservas = leerCI_('Reservas').filas, unidades = leerCI_('Unidades').filas;
  let cod = s_(d.codigo_reserva).toUpperCase();
  if (!cod && s_(d.id_visita)) {
    const v = leer_('Visitas', FILAS_RECIENTES).filas.find(x => s_(x.id_visita) === s_(d.id_visita).toUpperCase());
    if (v && s_(v.codigo_origen).indexOf('STL-') === 0) cod = s_(v.codigo_origen).toUpperCase();
  }
  const doc = normDoc_(d.documento);
  const activa = r => ['ingresado', 'pendiente'].indexOf(s_(r.estatus)) >= 0;
  let r = cod ? reservas.find(x => s_(x.codigo_reserva).toUpperCase() === cod) : null;
  if (!r && doc) {
    const per = leerCI_('Personas').filas.filter(p => normDoc_(p.numero_documento) === doc).map(p => s_(p.codigo_reserva).toUpperCase());
    r = reservas.filter(x => activa(x) && (per.indexOf(s_(x.codigo_reserva).toUpperCase()) >= 0 || normDoc_(x.documento_principal) === doc))
      .sort((a, b) => fecha_(b.llegada) - fecha_(a.llegada))[0] || null;
  }
  if (!r) return null;
  const u = torreDeUnidad_(unidades, r.codigo_unidad);
  if (u && torre && !mismaTorre_(u.torre, torre)) return null;
  return { r: r, u: u };
}
function reportarAlHost_(id, d, lobby) {
  const x = reservaDeHuesped_(d, lobby.torre);
  if (!x) return false;
  const n = respaldoNorma_(lobby.torre, TEMA_INCIDENCIA[s_(d.tipo)] || 'sanciones') || respaldoNorma_(lobby.torre, 'sanciones');
  agregarCI_('Reportes_Conducta', { id: id, fecha: new Date(), codigo_reserva: s_(x.r.codigo_reserva), codigo_unidad: s_(x.r.codigo_unidad),
    id_host: s_(x.r.id_host), torre: lobby.torre, apartamento: x.u ? x.u.apto : '', nombre: s_(d.nombre).toUpperCase() || s_(x.r.huesped_principal),
    documento: s_(d.documento).toUpperCase(), tipo: s_(d.tipo), gravedad: s_(d.gravedad) || 'leve', descripcion: s_(d.descripcion),
    norma: n ? n.cita + ': ' + n.texto : '', portero: s_(d.portero), enviado: 'NO' });
  return true;
}

// Lista negra única (Visitantes + huéspedes del Check-in). Solo el Gerente la cambia.
// ===========================================================================
function accionListaNegra_(req, admin) {
  const g = admin.rol === 'gerente', m = v => g ? s_(v) : enmascararDoc_(v);
  const mapa = {};
  leer_('Lista_Negra').filas.forEach(x => { const k = normDoc_(x.documento); if (!k) return;
    mapa[k] = { documento: m(x.documento), nombre: s_(x.nombre), motivo: s_(x.motivo), alcance: s_(x.torre) || 'TODAS', fecha: x.fecha, agregado_por: s_(x.agregado_por),
      activo: s_(x.estatus).toLowerCase() !== 'retirado', origen: ['visitantes'], clave: g ? s_(x.documento) : '' }; });
  const sol = [];
  if (libroCheckin_()) {
    leerCI_('Lista_Negra').filas.forEach(x => { const k = normDoc_(x.numero_documento); if (!k) return;
      const act = s_(x.estatus).toLowerCase() !== 'retirado';
      if (mapa[k]) { mapa[k].origen.push('huespedes'); mapa[k].activo = mapa[k].activo || act; if (!mapa[k].nombre) mapa[k].nombre = s_(x.nombre); }
      else mapa[k] = { documento: m(x.numero_documento), nombre: s_(x.nombre), motivo: s_(x.motivo), alcance: 'TODAS', fecha: x.fecha, agregado_por: s_(x.agregado_por),
        activo: act, origen: ['huespedes'], clave: g ? s_(x.numero_documento) : '' }; });
    leerCI_('Solicitudes_LN').filas.filter(x => s_(x.estatus) === 'pendiente').forEach(x => sol.push({ id: s_(x.id_solicitud), reserva: s_(x.codigo_reserva),
      documento: m(x.numero_documento), nombre: s_(x.nombre), host: s_(x.id_host), motivo: s_(x.motivo), fecha: x.fecha }));
  }
  const lista = Object.values(mapa).sort((a, b) => (b.activo - a.activo) || (fecha_(b.fecha) - fecha_(a.fecha)));
  return { lista: lista, solicitudes: sol, puede_editar: g };
}
function agregarListaNegra_(doc, nombre, motivo, torre, quien) {
  const d = normDoc_(doc);
  if (!d) throw error_('Escriba el número de documento.');
  if (!s_(motivo)) throw error_('Escriba el motivo.');
  const vis = leer_('Lista_Negra').filas.find(x => normDoc_(x.documento) === d);
  if (vis) actualizar_('Lista_Negra', vis._fila, { estatus: 'activo', motivo: s_(motivo), torre: s_(torre), nombre: s_(nombre).toUpperCase() || s_(vis.nombre), fecha: new Date(), agregado_por: quien });
  else agregar_('Lista_Negra', { documento: s_(doc).toUpperCase(), nombre: s_(nombre).toUpperCase(), motivo: s_(motivo), torre: s_(torre), fecha: new Date(), agregado_por: quien, estatus: 'activo' });
  if (libroCheckin_() && (!s_(torre) || s_(torre).toUpperCase() === 'TODAS')) {
    const ci = leerCI_('Lista_Negra').filas.find(x => normDoc_(x.numero_documento) === d);
    if (ci) actualizarCI_('Lista_Negra', ci._fila, { estatus: 'activo', motivo: s_(motivo) });
    else agregarCI_('Lista_Negra', { numero_documento: s_(doc).toUpperCase(), nombre: s_(nombre).toUpperCase(), motivo: s_(motivo), fecha: new Date(), agregado_por: quien, origen: 'stili', estatus: 'activo' });
  }
}
function accionListaNegraAgregar_(req, admin) {
  agregarListaNegra_(req.documento, req.nombre, req.motivo, req.torre, admin.nombre);
  // Si es empleado, queda inactivo
  const e = leer_('Empleados').filas.find(x => normDoc_(x.documento) === normDoc_(req.documento) && esSi_(x.activo));
  if (e) actualizar_('Empleados', e._fila, { activo: 'NO' });
  registrar_({ nombre: 'admin' }, admin.nombre, 'lista_negra_agregar', s_(req.documento) + ' · ' + s_(req.motivo));
  return Object.assign(accionListaNegra_(req, admin), { empleado_desactivado: !!e });
}
function accionListaNegraRetirar_(req, admin) {
  const d = normDoc_(req.documento);
  leer_('Lista_Negra').filas.filter(x => normDoc_(x.documento) === d).forEach(x => actualizar_('Lista_Negra', x._fila, { estatus: 'retirado' }));
  if (libroCheckin_()) leerCI_('Lista_Negra').filas.filter(x => normDoc_(x.numero_documento) === d).forEach(x => actualizarCI_('Lista_Negra', x._fila, { estatus: 'retirado' }));
  registrar_({ nombre: 'admin' }, admin.nombre, 'lista_negra_retirar', s_(req.documento));
  return accionListaNegra_(req, admin);
}
function accionSolicitudLNDecidir_(req, admin) {
  if (!libroCheckin_()) throw error_('Configure CHECKIN_SHEET_ID.');
  const s = leerCI_('Solicitudes_LN').filas.find(x => s_(x.id_solicitud) === s_(req.id));
  if (!s) throw error_('Solicitud no encontrada.');
  if (s_(s.estatus) !== 'pendiente') throw error_('Esa solicitud ya fue ' + s_(s.estatus) + '.');
  const acepta = req.decision === 'aceptar';
  actualizarCI_('Solicitudes_LN', s._fila, { estatus: acepta ? 'aceptada' : 'rechazada', revisado_por: admin.nombre });
  if (acepta) agregarListaNegra_(s.numero_documento, s.nombre, s_(s.motivo) || 'Solicitud del host ' + s_(s.id_host), 'TODAS', admin.nombre);
  registrar_({ nombre: 'admin' }, admin.nombre, 'solicitud_ln_' + (acepta ? 'aceptada' : 'rechazada'), s_(s.id_solicitud));
  return accionListaNegra_(req, admin);
}
/** Cada hora: copia al Check-in las normas y la lista negra general de Visitantes, para que el bot las aplique. */
function sincronizar() {
  try { sincronizarNormasCI_(); } catch (err) { Logger.log('Normas: ' + err.message); }
  if (!libroCheckin_()) return;
  const ci = leerCI_('Lista_Negra').filas;
  leer_('Lista_Negra').filas.forEach(x => {
    const d = normDoc_(x.documento); if (!d) return;
    const general = !s_(x.torre) || s_(x.torre).toUpperCase() === 'TODAS';
    const act = s_(x.estatus).toLowerCase() !== 'retirado';
    const c = ci.find(y => normDoc_(y.numero_documento) === d);
    if (act && general && !c) agregarCI_('Lista_Negra', { numero_documento: s_(x.documento).toUpperCase(), nombre: s_(x.nombre), motivo: s_(x.motivo), fecha: x.fecha || new Date(), agregado_por: s_(x.agregado_por), origen: 'visitantes', estatus: 'activo' });
    if (!act && c && s_(c.origen) === 'visitantes' && s_(c.estatus).toLowerCase() !== 'retirado') actualizarCI_('Lista_Negra', c._fila, { estatus: 'retirado' });
  });
}
